   149→   149→function SPEditorSheet({
   150→   150→  editor,
   151→   151→  data,
   152→   152→  onClose,
   153→   153→}: {
   154→   154→  editor: { node: SPNode | null; phase: string; parent: SPNode | null };
   155→   155→  data: LibraryData;
   156→   156→  onClose: () => void;
   157→   157→}) {
   158→   158→  const node = editor.node;
   159→   159→  const [name, setName] = useState(node?.name ?? "");
   160→   160→  const [description, setDescription] = useState(node?.description ?? "");
   161→   161→  const [parentId, setParentId] = useState<string | null>(editor.parent?.id ?? null);
   162→   162→  const [saving, startTransition] = useTransition();
   163→   163→  const { toast } = useToast();
   164→   164→  const router = useRouter();
   165→   165→
   166→   166→  const phaseMeta = PHASE_META[editor.phase] ?? PHASE_META.OFFENSIVE;
   167→   167→  const nodeHasChildren = node ? data.subprinciples.some((s) => s.parentId === node.id) : false;
   168→   168→
   169→   169→  // Raíces candidatas de la misma fase (excluye el propio nodo)
   170→   170→  const parentOptions = data.subprinciples.filter(
   171→   171→    (s) => s.phase === editor.phase && !s.parentId && s.id !== node?.id,
   172→   172→  );
   173→   173→
   174→   174→  function save() {
   175→   175→    startTransition(async () => {
   176→   176→      const res = await saveSubPrinciple({
   177→   177→        id: node?.id ?? "",
   178→   178→        phase: editor.phase,
   179→   179→        name,
   180→   180→        description: description.trim() || null,
   181→   181→        parentId: nodeHasChildren ? null : parentId,
   182→   182→      });
   183→   183→      if (res.ok) {
   184→   184→        toast({
   185→   185→          title: node ? "Subprincipio actualizado" : "Subprincipio creado en tu club",
   186→   186→          description: name,
   187→   187→          duration: 2000,
   188→   188→        });
   189→   189→        router.refresh();
   190→   190→        onClose();
   191→   191→      } else {
   192→   192→        toast({ title: "No se pudo guardar", description: res.error, duration: 5000 });
   193→   193→      }
   194→   194→    });
   195→   195→  }
   196→   196→
   197→   197→  return (
   198→   198→    <Sheet open onOpenChange={(o) => !o && onClose()}>
   199→   199→      <SheetContent
   200→   200→        side="bottom"
   201→   201→        className="bg-zinc-900 border-zinc-800 rounded-t-3xl max-h-[92vh] overflow-y-auto px-4 pb-6 pt-2"
   202→   202→      >
   203→   203→        <SheetHeader className="text-left pb-1">
   204→   204→          <SheetTitle className="text-xl font-black text-zinc-50">
   205→   205→            {node ? "Editar subprincipio" : editor.parent ? "Nuevo detalle" : "Nuevo subprincipio"}
   206→   206→          </SheetTitle>
   207→   207→          <SheetDescription className="flex items-center gap-1.5 text-sm">
   208→   208→            <span
   209→   209→              className={`inline-flex items-center gap-1 rounded-lg border px-2 py-0.5 text-[11px] font-black ${phaseMeta.chip}`}
   210→   210→            >
   211→   211→              {(() => {
   212→   212→                const Icon = phaseMeta.icon;
   213→   213→                return <Icon className="w-3.5 h-3.5" aria-hidden />;
   214→   214→              })()}
   215→   215→              {phaseMeta.label}
   216→   216→            </span>
   217→   217→            <span className="text-zinc-500">· la fase no se puede cambiar tras crear</span>
   218→   218→          </SheetDescription>
   219→   219→        </SheetHeader>
   220→   220→
   221→   221→        <Input
   222→   222→          value={name}
   223→   223→          onChange={(e) => setName(e.target.value)}
   224→   224→          placeholder="Nombre — ej. «Salida de balón desde presión rival en 3/4»"
   225→   225→          maxLength={120}
   226→   226→          className="mt-2 min-h-[48px] rounded-2xl bg-zinc-800/60 border-zinc-800 text-zinc-100 placeholder:text-zinc-600 font-semibold"
   227→   227→        />
   228→   228→        <Textarea
   229→   229→          value={description}
   230→   230→          onChange={(e) => setDescription(e.target.value)}
   231→   231→          placeholder="Descripción metodológica (opcional)"
   232→   232→          maxLength={500}
   233→   233→          rows={3}
   234→   234→          className="mt-2 rounded-2xl bg-zinc-800/60 border-zinc-800 text-zinc-100 placeholder:text-zinc-600"
   235→   235→        />
   236→   236→
   237→   237→        {/* Jerarquía (solo si el nodo no tiene hijos) */}
   238→   238→        {nodeHasChildren ? (
   239→   239→          <p className="mt-3 text-xs text-zinc-500">
   240→   240→            Este subprincipio tiene detalles propios: se mantiene como raíz de la jerarquía.
   241→   241→          </p>
   242→   242→        ) : (
   243→   243→          <>
   244→   244→            <p className="text-[10px] font-bold text-zinc-500 uppercase tracking-widest mt-4 mb-2">
   245→   245→              Jerarquía — ¿raíz o detalle de otro subprincipio?
   246→   246→            </p>
   247→   247→            <div className="flex flex-wrap gap-2">
   248→   248→              <button
   249→   249→                type="button"
   250→   250→                onClick={() => setParentId(null)}
   251→   251→                aria-pressed={parentId === null}
   252→   252→                className={`min-h-[44px] px-3.5 rounded-2xl border text-sm font-bold transition ${
   253→   253→                  parentId === null
   254→   254→                    ? "bg-emerald-500/15 border-emerald-500/50 text-emerald-300"
   255→   255→                    : "bg-zinc-800/60 border-zinc-800 text-zinc-400"
   256→   256→                }`}
   257→   257→              >
   258→   258→                Raíz de la fase
   259→   259→              </button>
   260→   260→              {parentOptions.map((p) => (
   261→   261→                <button
   262→   262→                  key={p.id}
   263→   263→                  type="button"
   264→   264→                  onClick={() => setParentId(p.id)}
   265→   265→                  aria-pressed={parentId === p.id}
   266→   266→                  className={`min-h-[44px] max-w-[48%] px-3.5 rounded-2xl border text-sm font-bold transition truncate ${
   267→   267→                    parentId === p.id
   268→   268→                      ? "bg-emerald-500/15 border-emerald-500/50 text-emerald-300"
   269→   269→                      : "bg-zinc-800/60 border-zinc-800 text-zinc-400"
   270→   270→                  }`}
   271→   271→                >
   272→   272→                  {p.name}
   273→   273→                </button>
   274→   274→              ))}
   275→   275→            </div>
   276→   276→          </>
   277→   277→        )}
   278→   278→
   279→   279→        <Button
   280→   280→          onClick={save}
   281→   281→          disabled={saving}
   282→   282→          className="mt-4 w-full min-h-[56px] rounded-2xl bg-emerald-500 hover:bg-emerald-400 text-zinc-950 font-black text-base"
   283→   283→        >
   284→   284→          {saving ? (
   285→   285→            <Loader2 className="w-5 h-5 animate-spin" aria-hidden />
   286→   286→          ) : (
   287→   287→            <>
   288→   288→              <PenLine className="w-5 h-5" aria-hidden />
   289→   289→              {node ? "Guardar cambios" : "Crear"}
   290→   290→            </>
   291→   291→          )}
   292→   292→        </Button>
   293→   293→      </SheetContent>
   294→   294→    </Sheet>
   295→   295→  );
   296→   296→}

   300→   300→export function TaxonomyTab({
   301→   301→  data,
   302→   302→  onOpenTask,
   303→   303→  onEditTask,
   304→   304→  onNewTask,
   305→   305→}: {
   306→   306→  data: LibraryData;
   307→   307→  onOpenTask: (t: TaskCard) => void;
   308→   308→  onEditTask: (t: TaskCard) => void;
   309→   309→  onNewTask: () => void;
   310→   310→}) {
   311→   311→  const [open, setOpen] = useState<Record<string, boolean>>(() =>
   312→   312→    Object.fromEntries(PHASE_ORDER.map((p) => [p, true])),
   313→   313→  );
   314→   314→  const [detail, setDetail] = useState<SPNode | null>(null);
   315→   315→  const [editor, setEditor] = useState<{
   316→   316→    node: SPNode | null;
   317→   317→    phase: string;
   318→   318→    parent: SPNode | null;
   319→   319→  } | null>(null);
   320→   320→  const [confirmDelete, setConfirmDelete] = useState(false);
   321→   321→  const [busy, startTransition] = useTransition();
   322→   322→  const { toast } = useToast();
   323→   323→  const router = useRouter();
   324→   324→
   325→   325→  const rootsOf = (phase: string) =>
   326→   326→    data.subprinciples
   327→   327→      .filter((s) => s.phase === phase && !s.parentId)
   328→   328→      .sort((a, b) => a.order - b.order || a.name.localeCompare(b.name));
   329→   329→  const childrenOf = (id: string) =>
   330→   330→    data.subprinciples.filter((s) => s.parentId === id).sort((a, b) => a.order - b.order);
   331→   331→  const recipeCountOf = (id: string) =>
   332→   332→    data.tasks.reduce((acc, t) => acc + (t.recipes.some((r) => r.subPrincipleId === id) ? 1 : 0), 0);
   333→   333→
   334→   334→  function runDuplicate(node: SPNode) {
   335→   335→    startTransition(async () => {
   336→   336→      const res = await duplicateSubPrincipleToClub(node.id);
   337→   337→      if (res.ok) {
   338→   338→        toast({
   339→   339→          title: "Subprincipio duplicado a tu club",
   340→   340→          description: "Ya puedes editarlo y anidar detalles propios.",
   341→   341→          duration: 2200,
   342→   342→        });
   343→   343→        router.refresh();
   344→   344→        setDetail(null);
   345→   345→      } else {
   346→   346→        toast({ title: "No se pudo duplicar", description: res.error, duration: 5000 });
   347→   347→      }
   348→   348→    });
   349→   349→  }
   350→   350→
   351→   351→  function runDelete() {
   352→   352→    if (!detail) return;
   353→   353→    const target = detail;
   354→   354→    startTransition(async () => {
   355→   355→      const res = await deleteSubPrinciple(target.id);
   356→   356→      if (res.ok) {
   357→   357→        toast({ title: "Subprincipio eliminado", description: target.name, duration: 2000 });
   358→   358→        router.refresh();
   359→   359→        setConfirmDelete(false);
   360→   360→        setDetail(null);
   361→   361→      } else {
   362→   362→        toast({ title: "No se pudo eliminar", description: res.error, duration: 5000 });
   363→   363→        setConfirmDelete(false);
   364→   364→      }
   365→   365→    });
   366→   366→  }
   367→   367→
   368→   368→  return (
   369→   369→    <div className="space-y-3">
   370→   370→      {PHASE_ORDER.map((phase) => {
   371→   371→        const meta = PHASE_META[phase];
   372→   372→        const Icon = meta.icon;
   373→   373→        const roots = rootsOf(phase);
   374→   374→        const phaseSPs = data.subprinciples.filter((s) => s.phase === phase);
   375→   375→        const hasIssue = phaseSPs.some((s) => s.issue);
   376→   376→        const expanded = open[phase] ?? true;
   377→   377→
   378→   378→        return (
   379→   379→          <section
   380→   380→            key={phase}
   381→   381→            aria-label={`Fase ${meta.label}`}
   382→   382→            className="rounded-3xl border border-zinc-800 bg-zinc-900/40 overflow-hidden"
   383→   383→          >
   384→   384→            {/* Cabecera de fase */}
   385→   385→            <div className="flex items-center gap-2 px-3 py-3">
   386→   386→              <button
   387→   387→                onClick={() => setOpen((o) => ({ ...o, [phase]: !expanded }))}
   388→   388→                className="flex-1 min-w-0 flex items-center gap-2.5 text-left min-h-[44px]"
   389→   389→                aria-expanded={expanded}
   390→   390→              >
   391→   391→                <span
   392→   392→                  className={`w-9 h-9 rounded-xl border flex items-center justify-center shrink-0 ${meta.chip}`}
   393→   393→                >
   394→   394→                  <Icon className="w-4.5 h-4.5" aria-hidden />
   395→   395→                </span>
   396→   396→                <span className="min-w-0">
   397→   397→                  <span className="block font-black text-zinc-50 text-[15px] leading-tight truncate">
   398→   398→                    {meta.label}
   399→   399→                    {hasIssue && (
   400→   400→                      <AlertTriangle className="w-4 h-4 text-rose-300 inline ml-1.5 -mt-0.5" aria-label="Con hallazgos abiertos esta semana" />
   401→   401→                    )}
   402→   402→                  </span>
   403→   403→                  <span className="block text-[11px] text-zinc-500 leading-tight truncate">
   404→   404→                    {roots.length} subprincipio{roots.length !== 1 ? "s" : ""} · {meta.desc}
   405→   405→                  </span>
   406→   406→                </span>
   407→   407→                <ChevronDown
   408→   408→                  className={`w-5 h-5 text-zinc-500 shrink-0 transition-transform ${expanded ? "" : "-rotate-90"}`}
   409→   409→                  aria-hidden
   410→   410→                />
   411→   411→              </button>
   412→   412→              <button
   413→   413→                onClick={() => setEditor({ node: null, phase, parent: null })}
   414→   414→                className="shrink-0 min-h-[44px] min-w-[44px] rounded-xl border border-emerald-500/40 bg-emerald-500/10 text-emerald-300 flex items-center justify-center hover:bg-emerald-500/20 transition"
   415→   415→                aria-label={`Añadir subprincipio a ${meta.label}`}
   416→   416→                title="Añadir subprincipio del club"
   417→   417→              >
   418→   418→                <Plus className="w-5 h-5" aria-hidden />
   419→   419→              </button>
   420→   420→            </div>
   421→   421→
   422→   422→            {/* Árbol */}
   423→   423→            {expanded && (
   424→   424→              <div className="px-3 pb-3 space-y-2">
   425→   425→                {roots.length === 0 && (
   426→   426→                  <p className="rounded-2xl border border-dashed border-zinc-800 px-4 py-4 text-center text-sm text-zinc-500">
   427→   427→                    Sin subprincipios en esta fase todavía.
   428→   428→                  </p>
   429→   429→                )}
   430→   430→                {roots.map((root) => {
   431→   431→                  const children = childrenOf(root.id);
   432→   432→                  return (
   433→   433→                    <div key={root.id} className="space-y-1.5">
   434→   434→                      <SPRow
   435→   435→                        node={root}
   436→   436→                        recipeCount={recipeCountOf(root.id)}
   437→   437→                        childCount={children.length}
   438→   438→                        onOpen={setDetail}
   439→   439→                      />
   440→   440→                      {children.map((c) => (
   441→   441→                        <div key={c.id} className="pl-5">

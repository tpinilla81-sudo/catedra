    36→/// Roles de la plataforma (seguridad y permisos)

    43→/// Fases del juego — taxonomía táctica de primer nivel
    44→enum GamePhase {
    45→  OFFENSIVE /// Organización ofensiva
    46→  DEFENSIVE /// Organización defensiva
    47→  OFF_TRANSITION /// Transición ofensiva (pérdida → reacción inmediata)
    48→  DEF_TRANSITION /// Transición defensiva (recuperación → progresión)
    49→  SET_PIECE /// Acciones a balón parado (ABP)
    50→}

    52→/// Demarcaciones (principal + polivalencias). Códigos internacionales estándar.
    53→enum Position {
    54→  GK /// Portero
    55→  CB /// Central
    56→  LB /// Lateral izquierdo
    57→  RB /// Lateral derecho
    58→  LWB /// Carrilero izquierdo
    59→  RWB /// Carrilero derecho
    60→  DM /// Pivot defensivo (MCD)
    61→  CM /// Medio centro
    62→  AM /// Mediapunta (MCO)
    63→  LM /// Interior / extremo retrasado izquierdo
    64→  RM /// Interior / extremo retrasado derecho
    65→  LW /// Extremo izquierdo
    66→  RW /// Extremo derecho
    67→  SS /// Segundo delantero (media punta alta)
    68→  CF /// Delantero centro (9)
    69→}

    71→/// Estado de disponibilidad del jugador (filtra la prescripción del motor)
    72→enum PlayerStatus {

    75→  INJURED
    76→  SUSPENDED
    77→  RETURNING /// Reincorporándose: el motor limita la intensidad de sus tareas

    87→/// Eventing táctico en la pizarra: marcado exprés por toques

   107→/// Ciclo de vida del partido

   114→/// Ciclo de vida del microciclo semanal ("cero pantallas en blanco")
   115→enum MicrocycleStatus {
   116→  PROPOSED /// Cocinado por el motor — el míster lo encuentra el lunes listo
   117→  ACCEPTED /// Validado con un toque
   118→  ADJUSTED /// Retocado a mano por el cuerpo técnico
   119→  COMPLETED /// Semana ejecutada y cerrada
   120→}
   121→
   122→/// Naturaleza del hallazgo del diagnóstico semanal
   123→enum IssueKind {
   124→  TACTICAL /// Falla táctica detectada en el eventing de la pizarra
   125→  PHYSICAL /// Fatiga / carga preocupante según Pulso y sliders
   126→  MORALE /// Inercia anímica en riesgo según Pulso y semáforo

   129→/// Estado del hallazgo (cierre del ciclo prescriptivo)

   376→// ============================================================================
   377→// PILAR 4.1 — SCOUTING DEL RIVAL (insumo del motor)
   378→// ----------------------------------------------------------------------------
   379→// Un informe por equipo y semana (clave teamId + weekStart, lunes-Madrid con
   380→// la MISMA convención de Microcycle.weekStart: medianoche UTC del lunes).
   381→// El motor lee el informe en collectInputs y convierte cada debilidad en un
   382→// hallazgo TACTICAL (source = OPPONENT_SCOUTING en la métrica de auditoría):
   383→// la semana no solo corrige lo propio, también entrena para EXPLOTAR al rival.
   384→// Los hijos cuelgan del informe (RELATION_SCOPED en el Tenant Guard).
   385→// ============================================================================
   386→
   387→/// Informe de scouting del rival de la semana — el que el motor cruza con el
   388→/// diagnóstico semanal. matchId es opcional: el informe puede existir sin
   389→/// partido agendado, y si el partido se borra el informe sobrevive (SetNull).
   390→model OpponentReport {
   391→  id             String   @id @default(cuid())
   392→  clubId         String
   393→  teamId         String
   394→  opponentName   String /// Nombre del rival estudiado
   395→  weekStart      DateTime @db.Date /// Lunes-Madrid de la semana (medianoche UTC)
   396→  matchId        String?  @unique /// Partido concreto asociado (opcional)
   397→  buildUpType    String /// Ej: "Salida lavolpiana, corta en corto"
   398→  defensiveBlock String /// Ej: "Bloque medio-bajo en 4-4-2"
   399→  createdAt      DateTime @default(now())
   400→  updatedAt      DateTime @updatedAt
   401→
   402→  weaknesses     OpponentWeakness[] /// Debilidades explotables (alimentan el motor)
   403→  scoutingEvents OpponentTacticalEvent[] /// Comportamientos observados (evidencia)
   404→
   405→  match Match? @relation(fields: [matchId], references: [id], onDelete: SetNull)
   406→  team  Team   @relation(fields: [teamId], references: [id], onDelete: Cascade)
   407→  club  Club   @relation(fields: [clubId], references: [id], onDelete: Cascade)
   408→
   409→  @@unique([teamId, weekStart]) /// Un informe por equipo y semana (como Microcycle)
   410→  @@index([clubId])
   411→}
   412→
   413→/// Debilidad detectada en el rival, conectada DIRECTAMENTE a la fase del

   416→model OpponentWeakness {

   419→  phase            GamePhase /// Fase donde el rival sufre (OFF_TRANSITION, ...)

   421→  severity         Int       @default(3) /// 1-5 (5 = muy explotable)
   422→
   423→  opponentReport OpponentReport @relation(fields: [opponentReportId], references: [id], onDelete: Cascade)
   424→
   425→  @@index([opponentReportId])
   426→}

   445→/// Taxonomía táctica jerárquica: fase → subprincipio → detalle (auto-relación).
   446→/// clubId = null → contenido oficial de la Cátedra (plataforma).
   447→/// clubId = X    → subprincipio privado del club.
   448→model SubPrinciple {
   449→  id          String    @id @default(cuid())
   450→  clubId      String? /// null = biblioteca global de la Cátedra
   451→  parentId    String? /// Jerarquía opcional: subprincipio → detalle
   452→  phase       GamePhase /// Fase del juego a la que pertenece
   453→  name        String /// "Salida de balón desde presión rival en 3/4"
   454→  description String?
   455→  order       Int       @default(0) /// Orden de presentación en la biblioteca
   456→  createdAt   DateTime  @default(now())
   457→
   458→  club     Club?              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   459→  parent   SubPrinciple?      @relation("SubPrincipleTree", fields: [parentId], references: [id], onDelete: SetNull)
   460→  children SubPrinciple[]     @relation("SubPrincipleTree")
   461→  tasks    TaskSubPrinciple[] /// Tareas conectadas ("recetas")
   462→  issues   DiagnosisIssue[] /// Hallazgos del motor que apuntan aquí
   463→
   464→  @@index([clubId, phase])
   465→  @@index([parentId])
   466→}

   468→/// Ficha de Tarea avanzada — la unidad de conocimiento entrenable.
   469→/// Conecta con la taxonomía vía TaskSubPrinciple ("recetas").
   470→model Task {
   471→  id                 String      @id @default(cuid())
   472→  clubId             String? /// null = biblioteca global de la Cátedra
   473→  createdById        String?
   474→  title              String
   475→  objective          String /// Qué se busca entrenar, en lenguaje de míster
   476→  phases             GamePhase[] /// Fases del juego que trabaja (una tarea puede tocar varias)
   477→  spaceLengthM       Int? /// Dimensiones del espacio: largo (m)
   478→  spaceWidthM        Int? /// Dimensiones del espacio: ancho (m)
   479→  spaceNote          String? /// "Con dos zonas laterales libres"
   480→  playersMin         Int /// Jugadores requeridos (mínimo)
   481→  playersMax         Int? /// Jugadores requeridos (máximo, si aplica)
   482→  durationMin        Int /// Tiempo total sugerido (min)
   483→  repetitions        String? /// "3 series x 6'", "2 bloques x 8'"
   484→  intensity          Intensity   @default(MEDIUM) /// El motor la cruza con la fatiga grupal
   485→  rulesOfProvocation String[] /// Reglas de provocación — array nativo de PostgreSQL
   486→  coachingPoints     String[] /// Puntos clave para el entrenador
   487→  materials          String[] /// Conos, petos, maniquíes...
   488→  diagramUrl         String? /// Diagrama / pizarra de la tarea
   489→  createdAt          DateTime    @default(now())
   490→  updatedAt          DateTime    @updatedAt
   491→
   492→  club          Club?              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   493→  createdBy     User?              @relation("TaskAuthor", fields: [createdById], references: [id], onDelete: SetNull)
   494→  subPrinciples TaskSubPrinciple[] /// Conexiones con la taxonomía ("recetas")
   495→  sessions      SessionTask[] /// Prescripciones históricas en microciclos
   496→
   497→  @@index([clubId])
   498→  @@index([intensity])
   499→}

   501→/// La "receta": conexión tarea ↔ subprincipio con relevancia ponderada.

   510→  createdAt      DateTime @default(now())
   511→
   512→  task         Task         @relation(fields: [taskId], references: [id], onDelete: Cascade)
   513→  subPrinciple SubPrinciple @relation(fields: [subPrincipleId], references: [id], onDelete: Cascade)
   514→
   515→  @@unique([taskId, subPrincipleId])
   516→  @@index([subPrincipleId, relevance])

   523→/// Microciclo semanal: la propuesta cocinada que el míster acepta con un toque.
   524→/// Unique (teamId, weekStart) garantiza que NUNCA haya pantallas en blanco.
   525→model Microcycle {
   526→  id                     String           @id @default(cuid())
   527→  clubId                 String
   528→  teamId                 String
   529→  weekStart              DateTime         @db.Date /// Lunes de la semana ISO
   530→  status                 MicrocycleStatus @default(PROPOSED)
   531→  tacticalDirectives     String[] /// Directrices tácticas de la semana
   532→  methodologicalApproach String? /// Enfoque metodológico global
   533→  diagnosisSummary       String? /// Resumen legible: qué falló + estado del grupo
   534→  engineVersion          String? /// Versión del algoritmo que generó la propuesta
   535→  acceptedById           String?
   536→  acceptedAt             DateTime?
   537→  createdAt              DateTime         @default(now())
   538→  updatedAt              DateTime         @updatedAt
   539→
   540→  club            Club              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   541→  team            Team              @relation(fields: [teamId], references: [id], onDelete: Cascade)
   542→  acceptedBy      User?             @relation("MicrocycleAccepter", fields: [acceptedById], references: [id], onDelete: SetNull)
   543→  analyzedMatches MicrocycleMatch[] /// Partidos cuya ingesta alimentó esta semana
   544→  issues          DiagnosisIssue[] /// Diagnóstico estructurado de la semana
   545→  sessions        Session[] /// Sesiones listas para el césped
   546→
   547→  @@unique([teamId, weekStart])
   548→  @@index([clubId, teamId, weekStart])
   549→}

   554→  matchId      String

   556→
   557→  microcycle Microcycle @relation(fields: [microcycleId], references: [id], onDelete: Cascade)

   564→/// Hallazgo del diagnóstico semanal con evidencia.
   565→/// Es el PUENTE entre los datos crudos (Pilar 3) y la prescripción (tareas).
   566→model DiagnosisIssue {
   567→  id             String      @id @default(cuid())
   568→  microcycleId   String
   569→  subPrincipleId String? /// Qué subprincipio falla (null = hallazgo físico/anímico)
   570→  phase          GamePhase? /// Fase donde se detecta el problema
   571→  kind           IssueKind
   572→  severity       Int         @default(2) /// 1-5 (5 = urgente) → prioriza la prescripción
   573→  evidence       String /// Evidencia legible: "9 pérdidas en salida por zona izquierda"
   574→  metric         Json? /// Números crudos del agregado (auditoría del motor)
   575→  status         IssueStatus @default(OPEN)
   576→  createdAt      DateTime    @default(now())
   577→
   578→  microcycle   Microcycle    @relation(fields: [microcycleId], references: [id], onDelete: Cascade)
   579→  subPrinciple SubPrinciple? @relation(fields: [subPrincipleId], references: [id], onDelete: SetNull)
   580→  prescribedIn SessionTask[] /// Tareas que lo abordan (trazabilidad receta → césped)
   581→
   582→  @@index([microcycleId])
   583→  @@index([subPrincipleId])
   584→}

   586→/// Sesión de entrenamiento dentro del microciclo (calendario MD+1, MD-3...)

   591→  label        String /// "MD+1", "MD-3", "MD-1" — código del calendario semanal
   592→  objective    String? /// Objetivo de la sesión en una frase
   593→  durationMin  Int?
   594→  createdAt    DateTime @default(now())
   595→
   596→  microcycle Microcycle    @relation(fields: [microcycleId], references: [id], onDelete: Cascade)
   597→  tasks      SessionTask[] /// Tareas exactas listas para el césped
   598→
   599→  @@index([microcycleId, date])
   600→}

   602→/// Tarea prescrita en una sesión — el punto final de la cadena del motor.
   603→/// `prescribedIssueId` da trazabilidad total: por qué esta tarea está aquí.
   604→model SessionTask {
   605→  id                String   @id @default(cuid())
   606→  sessionId         String
   607→  taskId            String
   608→  order             Int /// Orden dentro de la sesión (calentamiento, principal, cierre)
   609→  durationMin       Int? /// Tiempo real asignado a la tarea

   611→  prescribedIssueId String? /// Hallazgo que esta tarea ayuda a corregir

   613→
   614→  session Session         @relation(fields: [sessionId], references: [id], onDelete: Cascade)
   615→  task    Task            @relation(fields: [taskId], references: [id], onDelete: Cascade)
   616→  issue   DiagnosisIssue? @relation(fields: [prescribedIssueId], references: [id], onDelete: SetNull)
   617→
   618→  @@unique([sessionId, order])
   619→  @@index([taskId])
   620→}

   646→// ----------------------------------------------------------------------------
   647→// Extensión ADITIVA sobre el esquema de CATEDRA (no sustituye nada):
   648→//   • AttendanceRecord — quién ha venido y con qué RPE, marca de 1 toque.
   649→//   • LiveMatchEvaluation + MatchTag — tagging táctico en directo desde la
   650→//     pizarra mientras se mira el partido (positivos a reforzar / correctivos
   651→//     a corregir). El scope multi-tenant llega por relación:
   652→//     Player → Team/Club y Match → Team/Club.
   653→// ============================================================================

   655→/// Registro de asistencia e intensidad de una sesión (fútbol base).
   656→model AttendanceRecord {
   657→  id         String       @id @default(cuid())
   658→  date       DateTime     @default(now())
   659→  playerId   String
   660→  status     PlayerStatus @default(AVAILABLE)
   661→  sessionRPE Int? /// 1-10: intensidad marcada por el míster
   662→  notes      String?
   663→  createdAt  DateTime     @default(now())
   664→
   665→  player Player @relation(fields: [playerId], references: [id], onDelete: Cascade)
   666→
   667→  @@index([playerId, date])
   668→}
   669→
   670→/// Evaluación en directo de un jugador en un partido (1 por jugador/partido).
   671→model LiveMatchEvaluation {
   672→  id        String     @id @default(cuid())
   673→  matchId   String
   674→  playerId  String
   675→  isStarter Boolean    @default(false) /// Alineación de partida
   676→  timestamp DateTime   @default(now())
   677→  tags      MatchTag[]
   678→
   679→  match  Match  @relation(fields: [matchId], references: [id], onDelete: Cascade)
   680→  player Player @relation(fields: [playerId], references: [id], onDelete: Cascade)
   681→
   682→  @@unique([matchId, playerId])
   683→  @@index([matchId])
   684→}
   685→
   686→/// Tag táctico de 1 clic sobre una evaluación en directo.
   687→model MatchTag {
   688→  id           String   @id @default(cuid())
   689→  evaluationId String
   690→  code         TagCode
   691→  type         TagType
   692→  createdAt    DateTime @default(now())
   693→
   694→  evaluation LiveMatchEvaluation @relation(fields: [evaluationId], references: [id], onDelete: Cascade)
   695→
   696→  @@index([evaluationId, code])
   697→}

   699→/// Código del tag en directo: qué se ha visto en el jugador
   700→enum TagCode {
   701→  GOOD_UNMARKED /// Buen desmarque / profundidad
   702→  PASS_ACCURACY /// Precisión en el pase
   703→  DECISION_MAKING /// Buena toma de decisiones
   704→  TRACKBACK_QUICK /// Repliegue rápido
   705→  PRESS_INTENSITY /// Intensidad en la presión
   706→  POSITION_FAULT /// Descolgado de línea / Mala posición
   707→  OVER_DRIBBLING /// Exceso de conducción
   708→  NO_PRESS /// Falta de intensidad
   709→  FATIGUE_NOTICED /// Fatiga visible
   710→}
   711→
   712→/// Naturaleza del tag: a reforzar (positivo) o a corregir (correctivo)
   713→enum TagType {
   714→  POSITIVE
   715→  CORRECTIVE

    68→function SPRow({
    69→  node,
    70→  child,
    71→  recipeCount,
    72→  childCount,
    73→  onOpen,
    74→}: {
    75→  node: SPNode;
    76→  child?: boolean;
    77→  recipeCount: number;
    78→  childCount: number;
    79→  onOpen: (s: SPNode) => void;
    80→}) {
    81→  const origin = node.clubId === null ? ORIGIN_META.cathedral : ORIGIN_META.club;
    82→  const OriginIcon = origin.icon;
    83→
    84→  return (
    85→    <button
    86→      onClick={() => onOpen(node)}
    87→      className={`w-full rounded-2xl border text-left transition hover:border-zinc-600 active:scale-[0.99] focus-visible:outline focus-visible:outline-2 focus-visible:outline-emerald-400 ${
    88→        child
    89→          ? "border-zinc-800/70 bg-zinc-900/60"
    90→          : "border-zinc-800 bg-zinc-900"
    91→      }`}
    92→      aria-label={`Abrir subprincipio ${node.name}`}
    93→    >
    94→      <div className={`flex items-start gap-2 ${child ? "px-3 py-2.5" : "px-3.5 py-3"}`}>
    95→        {child && (
    96→          <CornerDownRight className="w-4 h-4 text-zinc-600 shrink-0 mt-0.5" aria-hidden />
    97→        )}
    98→        <div className="min-w-0 flex-1">
    99→          <div className="flex items-start justify-between gap-2">
   100→            <p className={`font-bold text-zinc-50 leading-snug ${child ? "text-sm" : "text-[15px]"}`}>
   101→              {node.name}
   102→            </p>
   103→            <span
   104→              className={`shrink-0 inline-flex items-center gap-1 rounded-lg border px-1.5 py-0.5 text-[10px] font-black ${origin.chip}`}
   105→              title={origin.hint}
   106→            >
   107→              <OriginIcon className="w-3 h-3" aria-hidden />
   108→              {origin.label}
   109→            </span>
   110→          </div>
   111→          {node.description && (
   112→            <p className="text-xs text-zinc-500 mt-0.5 line-clamp-1 leading-snug">
   113→              {node.description}
   114→            </p>
   115→          )}
   116→          <div className="flex flex-wrap items-center gap-1.5 mt-1.5">
   117→            {recipeCount > 0 ? (
   118→              <span className="rounded-md bg-amber-500/15 border border-amber-500/40 px-1.5 py-0.5 text-[10px] font-bold text-amber-300 inline-flex items-center gap-1">
   119→                <BookOpen className="w-3 h-3" aria-hidden /> {recipeCount} tarea
   120→                {recipeCount > 1 ? "s" : ""}
   121→              </span>
   122→            ) : (
   123→              <span className="rounded-md bg-zinc-800 px-1.5 py-0.5 text-[10px] font-bold text-zinc-500">
   124→                sin recetas
   125→              </span>
   126→            )}
   127→            {childCount > 0 && (
   128→              <span className="rounded-md bg-zinc-800 px-1.5 py-0.5 text-[10px] font-bold text-zinc-400">
   129→                {childCount} detalle{childCount > 1 ? "s" : ""}
   130→              </span>
   131→            )}
   132→            {node.issue && (
   133→              <span
   134→                className="rounded-md bg-rose-500/15 border border-rose-500/50 px-1.5 py-0.5 text-[10px] font-black text-rose-300 inline-flex items-center gap-1"
   135→                title={node.issue.evidence}
   136→              >
   137→                <AlertTriangle className="w-3 h-3" aria-hidden /> falla esta semana
   138→              </span>
   139→            )}
   140→          </div>
   141→        </div>
   142→      </div>
   143→    </button>
   144→  );
   145→}

   147→// ---------- Editor de subprincipio (crear / editar) ----------

   149→function SPEditorSheet({
   150→  editor,
   151→  data,
   152→  onClose,
   153→}: {
   154→  editor: { node: SPNode | null; phase: string; parent: SPNode | null };
   155→  data: LibraryData;
   156→  onClose: () => void;
   157→}) {
   158→  const node = editor.node;
   159→  const [name, setName] = useState(node?.name ?? "");
   160→  const [description, setDescription] = useState(node?.description ?? "");
   161→  const [parentId, setParentId] = useState<string | null>(editor.parent?.id ?? null);
   162→  const [saving, startTransition] = useTransition();
   163→  const { toast } = useToast();
   164→  const router = useRouter();
   165→
   166→  const phaseMeta = PHASE_META[editor.phase] ?? PHASE_META.OFFENSIVE;
   167→  const nodeHasChildren = node ? data.subprinciples.some((s) => s.parentId === node.id) : false;
   168→
   169→  // Raíces candidatas de la misma fase (excluye el propio nodo)
   170→  const parentOptions = data.subprinciples.filter(
   171→    (s) => s.phase === editor.phase && !s.parentId && s.id !== node?.id,
   172→  );
   173→
   174→  function save() {
   175→    startTransition(async () => {
   176→      const res = await saveSubPrinciple({
   177→        id: node?.id ?? "",
   178→        phase: editor.phase,
   179→        name,
   180→        description: description.trim() || null,
   181→        parentId: nodeHasChildren ? null : parentId,
   182→      });
   183→      if (res.ok) {
   184→        toast({
   185→          title: node ? "Subprincipio actualizado" : "Subprincipio creado en tu club",
   186→          description: name,
   187→          duration: 2000,
   188→        });
   189→        router.refresh();
   190→        onClose();
   191→      } else {
   192→        toast({ title: "No se pudo guardar", description: res.error, duration: 5000 });
   193→      }
   194→    });
   195→  }
   196→
   197→  return (
   198→    <Sheet open onOpenChange={(o) => !o && onClose()}>
   199→      <SheetContent
   200→        side="bottom"
   201→        className="bg-zinc-900 border-zinc-800 rounded-t-3xl max-h-[92vh] overflow-y-auto px-4 pb-6 pt-2"
   202→      >
   203→        <SheetHeader className="text-left pb-1">
   204→          <SheetTitle className="text-xl font-black text-zinc-50">
   205→            {node ? "Editar subprincipio" : editor.parent ? "Nuevo detalle" : "Nuevo subprincipio"}
   206→          </SheetTitle>
   207→          <SheetDescription className="flex items-center gap-1.5 text-sm">
   208→            <span
   209→              className={`inline-flex items-center gap-1 rounded-lg border px-2 py-0.5 text-[11px] font-black ${phaseMeta.chip}`}
   210→            >
   211→              {(() => {
   212→                const Icon = phaseMeta.icon;
   213→                return <Icon className="w-3.5 h-3.5" aria-hidden />;
   214→              })()}
   215→              {phaseMeta.label}
   216→            </span>
   217→            <span className="text-zinc-500">· la fase no se puede cambiar tras crear</span>
   218→          </SheetDescription>
   219→        </SheetHeader>
   220→
   221→        <Input
   222→          value={name}
   223→          onChange={(e) => setName(e.target.value)}
   224→          placeholder="Nombre — ej. «Salida de balón desde presión rival en 3/4»"
   225→          maxLength={120}
   226→          className="mt-2 min-h-[48px] rounded-2xl bg-zinc-800/60 border-zinc-800 text-zinc-100 placeholder:text-zinc-600 font-semibold"
   227→        />
   228→        <Textarea
   229→          value={description}
   230→          onChange={(e) => setDescription(e.target.value)}
   231→          placeholder="Descripción metodológica (opcional)"
   232→          maxLength={500}
   233→          rows={3}
   234→          className="mt-2 rounded-2xl bg-zinc-800/60 border-zinc-800 text-zinc-100 placeholder:text-zinc-600"
   235→        />
   236→
   237→        {/* Jerarquía (solo si el nodo no tiene hijos) */}
   238→        {nodeHasChildren ? (
   239→          <p className="mt-3 text-xs text-zinc-500">
   240→            Este subprincipio tiene detalles propios: se mantiene como raíz de la jerarquía.
   241→          </p>
   242→        ) : (
   243→          <>
   244→            <p className="text-[10px] font-bold text-zinc-500 uppercase tracking-widest mt-4 mb-2">
   245→              Jerarquía — ¿raíz o detalle de otro subprincipio?
   246→            </p>
   247→            <div className="flex flex-wrap gap-2">
   248→              <button
   249→                type="button"
   250→                onClick={() => setParentId(null)}
   251→                aria-pressed={parentId === null}
   252→                className={`min-h-[44px] px-3.5 rounded-2xl border text-sm font-bold transition ${
   253→                  parentId === null
   254→                    ? "bg-emerald-500/15 border-emerald-500/50 text-emerald-300"
   255→                    : "bg-zinc-800/60 border-zinc-800 text-zinc-400"
   256→                }`}
   257→              >
   258→                Raíz de la fase
   259→              </button>
   260→              {parentOptions.map((p) => (
   261→                <button
   262→                  key={p.id}
   263→                  type="button"
   264→                  onClick={() => setParentId(p.id)}
   265→                  aria-pressed={parentId === p.id}
   266→                  className={`min-h-[44px] max-w-[48%] px-3.5 rounded-2xl border text-sm font-bold transition truncate ${
   267→                    parentId === p.id
   268→                      ? "bg-emerald-500/15 border-emerald-500/50 text-emerald-300"
   269→                      : "bg-zinc-800/60 border-zinc-800 text-zinc-400"
   270→                  }`}
   271→                >
   272→                  {p.name}
   273→                </button>
   274→              ))}
   275→            </div>
   276→          </>
   277→        )}
   278→
   279→        <Button
   280→          onClick={save}
   281→          disabled={saving}
   282→          className="mt-4 w-full min-h-[56px] rounded-2xl bg-emerald-500 hover:bg-emerald-400 text-zinc-950 font-black text-base"
   283→        >
   284→          {saving ? (
   285→            <Loader2 className="w-5 h-5 animate-spin" aria-hidden />
   286→          ) : (
   287→            <>
   288→              <PenLine className="w-5 h-5" aria-hidden />
   289→              {node ? "Guardar cambios" : "Crear"}
   290→            </>
   291→          )}
   292→        </Button>
   293→      </SheetContent>
   294→    </Sheet>
   295→  );
   296→}

   298→// ---------- Pestaña principal ----------

   300→export function TaxonomyTab({
   301→  data,
   302→  onOpenTask,
   303→  onEditTask,
   304→  onNewTask,
   305→}: {
   306→  data: LibraryData;
   307→  onOpenTask: (t: TaskCard) => void;
   308→  onEditTask: (t: TaskCard) => void;
   309→  onNewTask: () => void;
   310→}) {
   311→  const [open, setOpen] = useState<Record<string, boolean>>(() =>
   312→    Object.fromEntries(PHASE_ORDER.map((p) => [p, true])),
   313→  );
   314→  const [detail, setDetail] = useState<SPNode | null>(null);
   315→  const [editor, setEditor] = useState<{
   316→    node: SPNode | null;
   317→    phase: string;
   318→    parent: SPNode | null;
   319→  } | null>(null);
   320→  const [confirmDelete, setConfirmDelete] = useState(false);
   321→  const [busy, startTransition] = useTransition();
   322→  const { toast } = useToast();
   323→  const router = useRouter();
   324→
   325→  const rootsOf = (phase: string) =>
   326→    data.subprinciples
   327→      .filter((s) => s.phase === phase && !s.parentId)
   328→      .sort((a, b) => a.order - b.order || a.name.localeCompare(b.name));
   329→  const childrenOf = (id: string) =>
   330→    data.subprinciples.filter((s) => s.parentId === id).sort((a, b) => a.order - b.order);
   331→  const recipeCountOf = (id: string) =>
   332→    data.tasks.reduce((acc, t) => acc + (t.recipes.some((r) => r.subPrincipleId === id) ? 1 : 0), 0);
   333→
   334→  function runDuplicate(node: SPNode) {
   335→    startTransition(async () => {
   336→      const res = await duplicateSubPrincipleToClub(node.id);
   337→      if (res.ok) {
   338→        toast({
   339→          title: "Subprincipio duplicado a tu club",
   340→          description: "Ya puedes editarlo y anidar detalles propios.",
   341→          duration: 2200,
   342→        });
   343→        router.refresh();
   344→        setDetail(null);
   345→      } else {
   346→        toast({ title: "No se pudo duplicar", description: res.error, duration: 5000 });
   347→      }
   348→    });
   349→  }
   350→
   351→  function runDelete() {
   352→    if (!detail) return;
   353→    const target = detail;
   354→    startTransition(async () => {
   355→      const res = await deleteSubPrinciple(target.id);
   356→      if (res.ok) {
   357→        toast({ title: "Subprincipio eliminado", description: target.name, duration: 2000 });
   358→        router.refresh();
   359→        setConfirmDelete(false);
   360→        setDetail(null);
   361→      } else {
   362→        toast({ title: "No se pudo eliminar", description: res.error, duration: 5000 });
   363→        setConfirmDelete(false);
   364→      }
   365→    });
   366→  }
   367→
   368→  return (
   369→    <div className="space-y-3">
   370→      {PHASE_ORDER.map((phase) => {
   371→        const meta = PHASE_META[phase];
   372→        const Icon = meta.icon;
   373→        const roots = rootsOf(phase);
   374→        const phaseSPs = data.subprinciples.filter((s) => s.phase === phase);
   375→        const hasIssue = phaseSPs.some((s) => s.issue);
   376→        const expanded = open[phase] ?? true;
   377→
   378→        return (
   379→          <section
   380→            key={phase}
   381→            aria-label={`Fase ${meta.label}`}
   382→            className="rounded-3xl border border-zinc-800 bg-zinc-900/40 overflow-hidden"
   383→          >
   384→            {/* Cabecera de fase */}
   385→            <div className="flex items-center gap-2 px-3 py-3">
   386→              <button
   387→                onClick={() => setOpen((o) => ({ ...o, [phase]: !expanded }))}
   388→                className="flex-1 min-w-0 flex items-center gap-2.5 text-left min-h-[44px]"
   389→                aria-expanded={expanded}
   390→              >
   391→                <span
   392→                  className={`w-9 h-9 rounded-xl border flex items-center justify-center shrink-0 ${meta.chip}`}
   393→                >
   394→                  <Icon className="w-4.5 h-4.5" aria-hidden />
   395→                </span>
   396→                <span className="min-w-0">
   397→                  <span className="block font-black text-zinc-50 text-[15px] leading-tight truncate">
   398→                    {meta.label}
   399→                    {hasIssue && (
   400→                      <AlertTriangle className="w-4 h-4 text-rose-300 inline ml-1.5 -mt-0.5" aria-label="Con hallazgos abiertos esta semana" />
   401→                    )}
   402→                  </span>
   403→                  <span className="block text-[11px] text-zinc-500 leading-tight truncate">
   404→                    {roots.length} subprincipio{roots.length !== 1 ? "s" : ""} · {meta.desc}
   405→                  </span>
   406→                </span>
   407→                <ChevronDown
   408→                  className={`w-5 h-5 text-zinc-500 shrink-0 transition-transform ${expanded ? "" : "-rotate-90"}`}
   409→                  aria-hidden
   410→                />
   411→              </button>
   412→              <button
   413→                onClick={() => setEditor({ node: null, phase, parent: null })}
   414→                className="shrink-0 min-h-[44px] min-w-[44px] rounded-xl border border-emerald-500/40 bg-emerald-500/10 text-emerald-300 flex items-center justify-center hover:bg-emerald-500/20 transition"
   415→                aria-label={`Añadir subprincipio a ${meta.label}`}
   416→                title="Añadir subprincipio del club"
   417→              >
   418→                <Plus className="w-5 h-5" aria-hidden />
   419→              </button>
   420→            </div>
   421→
   422→            {/* Árbol */}
   423→            {expanded && (
   424→              <div className="px-3 pb-3 space-y-2">
   425→                {roots.length === 0 && (
   426→                  <p className="rounded-2xl border border-dashed border-zinc-800 px-4 py-4 text-center text-sm text-zinc-500">
   427→                    Sin subprincipios en esta fase todavía.
   428→                  </p>
   429→                )}
   430→                {roots.map((root) => {
   431→                  const children = childrenOf(root.id);
   432→                  return (
   433→                    <div key={root.id} className="space-y-1.5">
   434→                      <SPRow
   435→                        node={root}
   436→                        recipeCount={recipeCountOf(root.id)}
   437→                        childCount={children.length}
   438→                        onOpen={setDetail}
   439→                      />
   440→                      {children.map((c) => (
   441→                        <div key={c.id} className="pl-5">
   442→                          <SPRow
   443→                            node={c}
   444→                            child
   445→                            recipeCount={recipeCountOf(c.id)}
   446→                            childCount={0}
   447→                            onOpen={setDetail}
   448→                          />
   449→                        </div>
   450→                      ))}
   451→                    </div>
   452→                  );
   453→                })}
   454→              </div>
   455→            )}
   456→          </section>
   457→        );
   458→      })}
   459→
   460→      {/* ---------- Hoja de detalle del subprincipio ---------- */}
   461→      <Sheet open={!!detail} onOpenChange={(o) => !o && setDetail(null)}>
   462→        <SheetContent
   463→          side="bottom"
   464→          className="bg-zinc-900 border-zinc-800 rounded-t-3xl max-h-[92vh] overflow-y-auto px-4 pb-8 pt-2"
   465→        >
   466→          {detail && (
   467→            <>
   468→              <SheetHeader className="text-left pb-1">
   469→                <SheetTitle className="text-xl font-black text-zinc-50 leading-snug">
   470→                  {detail.name}
   471→                </SheetTitle>
   472→                <SheetDescription className="flex flex-wrap items-center gap-1.5 pt-1">
   473→                  {(() => {
   474→                    const origin = detail.clubId === null ? ORIGIN_META.cathedral : ORIGIN_META.club;
   475→                    const OriginIcon = origin.icon;
   476→                    return (
   477→                      <span
   478→                        className={`inline-flex items-center gap-1 rounded-lg border px-2 py-0.5 text-[11px] font-black ${origin.chip}`}
   479→                        title={origin.hint}
   480→                      >
   481→                        <OriginIcon className="w-3.5 h-3.5" aria-hidden /> {origin.label}
   482→                      </span>
   483→                    );
   484→                  })()}
   485→                  {(() => {
   486→                    const pMeta = PHASE_META[detail.phase] ?? PHASE_META.OFFENSIVE;
   487→                    const Icon = pMeta.icon;
   488→                    return (
   489→                      <span
   490→                        className={`inline-flex items-center gap-1 rounded-lg border px-2 py-0.5 text-[11px] font-black ${pMeta.chip}`}
   491→                      >
   492→                        <Icon className="w-3.5 h-3.5" aria-hidden /> {pMeta.label}
   493→                      </span>
   494→                    );
   495→                  })()}
   496→                </SheetDescription>
   497→              </SheetHeader>
   498→
   499→              {detail.description && (
   500→                <p className="text-sm text-zinc-300 leading-relaxed mt-1">{detail.description}</p>
   501→              )}
   502→
   503→              {/* Hallazgo del motor */}
   504→              {detail.issue && (
   505→                <div className="mt-3 rounded-2xl bg-rose-500/10 border border-rose-500/40 px-3.5 py-3 flex gap-2.5">
   506→                  <AlertTriangle className="w-5 h-5 text-rose-300 shrink-0 mt-0.5" aria-hidden />
   507→                  <div>
   508→                    <p className="text-sm font-black text-rose-300">
   509→                      Falla esta semana (severidad {detail.issue.severity}/5)
   510→                    </p>
   511→                    <p className="text-sm text-rose-200/90 mt-0.5 leading-snug">
   512→                      {detail.issue.evidence}
   513→                    </p>
   514→                    <p className="text-[11px] text-zinc-500 mt-1">
   515→                      Las tareas con receta núcleo hacia aquí son la primera opción del motor para
   516→                      el lunes.
   517→                    </p>
   518→                  </div>
   519→                </div>
   520→              )}
   521→
   522→              {/* Detalles (hijos) */}
   523→              {childrenOf(detail.id).length > 0 && (
   524→                <>
   525→                  <p className="text-[10px] font-bold text-zinc-500 uppercase tracking-widest mt-5 mb-2">
   526→                    Detalles del subprincipio
   527→                  </p>
   528→                  <div className="space-y-1.5">
   529→                    {childrenOf(detail.id).map((c) => (
   530→                      <SPRow
   531→                        key={c.id}
   532→                        node={c}
   533→                        child
   534→                        recipeCount={recipeCountOf(c.id)}
   535→                        childCount={0}
   536→                        onOpen={setDetail}
   537→                      />
   538→                    ))}
   539→                  </div>
   540→                </>
   541→              )}
   542→
   543→              {/* RECETAS: tareas que lo resuelven (motor relacional bidireccional) */}
   544→              <p className="text-[10px] font-bold text-zinc-500 uppercase tracking-widest mt-5 mb-2">
   545→                Tareas que lo resuelven (recetas del motor)
   546→              </p>
   547→              {(() => {
   548→                const links = tasksForSubPrinciple(data.tasks, detail.id);
   549→                if (links.length === 0)
   550→                  return (
   551→                    <p className="rounded-2xl border border-dashed border-zinc-700 px-4 py-3 text-sm text-zinc-500">
   552→                      Ninguna tarea conectada todavía. Abre una tarea de tu club y añade una receta
   553→                      hacia este subprincipio — o crea una nueva.
   554→                    </p>
   555→                  );
   556→                return (
   557→                  <ul className="space-y-2">
   558→                    {links.map(({ task, relevance, usageNote }) => {
   559→                      const rel = RELEVANCE_META[relevance] ?? RELEVANCE_META[2];
   560→                      const intensity = INTENSITY_META[task.intensity] ?? INTENSITY_META.MEDIUM;
   561→                      return (
   562→                        <li key={task.id}>
   563→                          <button
   564→                            onClick={() => onOpenTask(task)}
   565→                            className="w-full rounded-2xl bg-zinc-800/50 border border-zinc-800 px-3 py-2.5 text-left hover:border-zinc-600 transition"
   566→                            aria-label={`Abrir tarea ${task.title}`}
   567→                          >
   568→                            <div className="flex items-center gap-2 flex-wrap">
   569→                              <span
   570→                                className={`rounded-md border px-1.5 py-0.5 text-[10px] font-black ${rel.chip}`}
   571→                                title={rel.hint}
   572→                              >
   573→                                {relevance} · {rel.label}
   574→                              </span>
   575→                              <p className="text-sm font-bold text-zinc-100 flex-1 min-w-[50%] leading-snug">
   576→                                {task.title}
   577→                              </p>
   578→                              <span
   579→                                className={`rounded-md border px-1.5 py-0.5 text-[10px] font-bold inline-flex items-center gap-1 ${intensity.chip}`}
   580→                              >
   581→                                <span className={`w-1.5 h-1.5 rounded-full ${intensity.dot}`} aria-hidden />
   582→                                {intensity.label}
   583→                              </span>
   584→                            </div>
   585→                            {usageNote && (
   586→                              <p className="text-xs italic text-zinc-400 mt-1.5 leading-snug">
   587→                                «{usageNote}»
   588→                              </p>
   589→                            )}
   590→                          </button>
   591→                        </li>
   592→                      );
   593→                    })}
   594→                  </ul>
   595→                );
   596→              })()}
   597→
   598→              {/* Acciones */}
   599→              <div className="mt-5 space-y-2">
   600→                {detail.clubId !== null ? (
   601→                  <>
   602→                    <div className="grid grid-cols-2 gap-2">
   603→                      <Button
   604→                        onClick={() =>
   605→                          setEditor({
   606→                            node: detail,
   607→                            phase: detail.phase,
   608→                            parent: null,
   609→                          })
   610→                        }
   611→                        className="min-h-[52px] rounded-2xl bg-emerald-500 hover:bg-emerald-400 text-zinc-950 font-black"
   612→                      >
   613→                        <PenLine className="w-5 h-5" aria-hidden /> Editar
   614→                      </Button>
   615→                      <Button
   616→                        onClick={() => setEditor({ node: null, phase: detail.phase, parent: detail })}
   617→                        className="min-h-[52px] rounded-2xl bg-zinc-800 hover:bg-zinc-700 text-zinc-100 font-black border border-zinc-700"
   618→                      >
   619→                        <Plus className="w-5 h-5" aria-hidden /> Detalle
   620→                      </Button>
   621→                    </div>
   622→                    <Button
   623→                      variant="outline"
   624→                      onClick={() => setConfirmDelete(true)}
   625→                      disabled={busy}
   626→                      className="w-full min-h-[48px] rounded-2xl border-zinc-700 bg-zinc-950 text-rose-300 hover:bg-rose-500/10 hover:text-rose-300 font-bold"
   627→                    >
   628→                      <Trash2 className="w-4 h-4" aria-hidden /> Eliminar subprincipio
   629→                    </Button>
   630→                  </>
   631→                ) : (
   632→                  <Button
   633→                    onClick={() => runDuplicate(detail)}
   634→                    disabled={busy}
   635→                    className="w-full min-h-[52px] rounded-2xl bg-violet-500/90 hover:bg-violet-400 text-zinc-950 font-black"
   636→                  >
   637→                    {busy ? (
   638→                      <Loader2 className="w-5 h-5 animate-spin" aria-hidden />
   639→                    ) : (
   640→                      <>
   641→                        <CopyPlus className="w-5 h-5" aria-hidden /> Duplicar a mi club para
   642→                        personalizar
   643→                      </>
   644→                    )}
   645→                  </Button>
   646→                )}
   647→                <Button
   648→                  variant="ghost"
   649→                  onClick={onNewTask}
   650→                  className="w-full min-h-[44px] rounded-2xl text-zinc-400 hover:text-zinc-200 font-bold"
   651→                >
   652→                  <BookOpen className="w-4 h-4" aria-hidden /> Crear una tarea para este objetivo
   653→                </Button>
   654→              </div>
   655→
   656→              <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   657→                <AlertDialogContent className="bg-zinc-900 border-zinc-800 rounded-3xl max-w-sm mx-auto">
   658→                  <AlertDialogHeader>
   659→                    <AlertDialogTitle className="text-zinc-50">
   660→                      ¿Eliminar «{detail.name}»?
   661→                    </AlertDialogTitle>
   662→                    <AlertDialogDescription className="text-zinc-400">
   663→                      Sus detalles quedarán como raíces y las recetas conectadas se desconectarán.
   664→                      Los hallazgos históricos del motor conservan su evidencia.
   665→                    </AlertDialogDescription>
   666→                  </AlertDialogHeader>
   667→                  <AlertDialogFooter className="flex-row gap-2">
   668→                    <AlertDialogCancel className="rounded-2xl border-zinc-700 bg-zinc-950 text-zinc-200 mt-0">
   669→                      Cancelar
   670→                    </AlertDialogCancel>
   671→                    <AlertDialogAction
   672→                      onClick={(e) => {
   673→                        e.preventDefault();
   674→                        runDelete();
   675→                      }}
   676→                      className="rounded-2xl bg-rose-500 text-white hover:bg-rose-400 mt-0"
   677→                    >
   678→                      {busy ? <Loader2 className="w-4 h-4 animate-spin" aria-hidden /> : "Eliminar"}
   679→                    </AlertDialogAction>
   680→                  </AlertDialogFooter>
   681→                </AlertDialogContent>
   682→              </AlertDialog>
   683→            </>
   684→          )}
   685→        </SheetContent>
   686→      </Sheet>
   687→
   688→      {/* ---------- Editor ---------- */}
   689→      {editor && (
   690→        <SPEditorSheet editor={editor} data={data} onClose={() => setEditor(null)} />
   691→      )}
   692→    </div>
   693→  );
   694→}

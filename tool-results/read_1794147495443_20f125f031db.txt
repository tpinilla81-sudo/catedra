    42→/// Fases del juego — taxonomía táctica de primer nivel
    43→enum GamePhase {
    44→  OFFENSIVE /// Organización ofensiva
    45→  DEFENSIVE /// Organización defensiva

    51→/// Demarcaciones (principal + polivalencias). Códigos internacionales estándar.

    58→  RWB /// Carrilero derecho
    59→  DM /// Pivot defensivo (MCD)
    60→  CM /// Medio centro
    61→  AM /// Mediapunta (MCO)

    63→  RM /// Interior / extremo retrasado derecho
    64→  LW /// Extremo izquierdo
    65→  RW /// Extremo derecho
    66→  SS /// Segundo delantero (media punta alta)
    67→  CF /// Delantero centro (9)

   112→/// Ciclo de vida del microciclo semanal ("cero pantallas en blanco")

   120→/// Naturaleza del hallazgo del diagnóstico semanal

   371→// ============================================================================
   372→// PILAR 4.1 — SCOUTING DEL RIVAL (insumo del motor)
   373→// ----------------------------------------------------------------------------
   374→// Un informe por equipo y semana (clave teamId + weekStart, lunes-Madrid con
   375→// la MISMA convención de Microcycle.weekStart: medianoche UTC del lunes).
   376→// El motor lee el informe en collectInputs y convierte cada debilidad en un
   377→// hallazgo TACTICAL (source = OPPONENT_SCOUTING en la métrica de auditoría):
   378→// la semana no solo corrige lo propio, también entrena para EXPLOTAR al rival.
   379→// Los hijos cuelgan del informe (RELATION_SCOPED en el Tenant Guard).
   380→// ============================================================================
   381→
   382→/// Informe de scouting del rival de la semana — el que el motor cruza con el
   383→/// diagnóstico semanal. matchId es opcional: el informe puede existir sin
   384→/// partido agendado, y si el partido se borra el informe sobrevive (SetNull).
   385→model OpponentReport {
   386→  id             String   @id @default(cuid())
   387→  clubId         String
   388→  teamId         String
   389→  opponentName   String /// Nombre del rival estudiado
   390→  weekStart      DateTime @db.Date /// Lunes-Madrid de la semana (medianoche UTC)
   391→  matchId        String?  @unique /// Partido concreto asociado (opcional)
   392→  buildUpType    String /// Ej: "Salida lavolpiana, corta en corto"
   393→  defensiveBlock String /// Ej: "Bloque medio-bajo en 4-4-2"
   394→  createdAt      DateTime @default(now())
   395→  updatedAt      DateTime @updatedAt
   396→
   397→  weaknesses     OpponentWeakness[] /// Debilidades explotables (alimentan el motor)
   398→  scoutingEvents OpponentTacticalEvent[] /// Comportamientos observados (evidencia)
   399→
   400→  match Match? @relation(fields: [matchId], references: [id], onDelete: SetNull)
   401→  team  Team   @relation(fields: [teamId], references: [id], onDelete: Cascade)
   402→  club  Club   @relation(fields: [clubId], references: [id], onDelete: Cascade)
   403→
   404→  @@unique([teamId, weekStart]) /// Un informe por equipo y semana (como Microcycle)
   405→  @@index([clubId])
   406→}
   407→
   408→/// Debilidad detectada en el rival, conectada DIRECTAMENTE a la fase del

   411→model OpponentWeakness {

   416→  severity         Int       @default(3) /// 1-5 (5 = muy explotable)
   417→
   418→  opponentReport OpponentReport @relation(fields: [opponentReportId], references: [id], onDelete: Cascade)
   419→
   420→  @@index([opponentReportId])
   421→}

   440→/// Taxonomía táctica jerárquica: fase → subprincipio → detalle (auto-relación).

   442→/// clubId = X    → subprincipio privado del club.
   443→model SubPrinciple {

   446→  parentId    String? /// Jerarquía opcional: subprincipio → detalle

   448→  name        String /// "Salida de balón desde presión rival en 3/4"
   449→  description String?
   450→  order       Int       @default(0) /// Orden de presentación en la biblioteca
   451→  createdAt   DateTime  @default(now())
   452→
   453→  club     Club?              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   454→  parent   SubPrinciple?      @relation("SubPrincipleTree", fields: [parentId], references: [id], onDelete: SetNull)
   455→  children SubPrinciple[]     @relation("SubPrincipleTree")
   456→  tasks    TaskSubPrinciple[] /// Tareas conectadas ("recetas")
   457→  issues   DiagnosisIssue[] /// Hallazgos del motor que apuntan aquí
   458→
   459→  @@index([clubId, phase])
   460→  @@index([parentId])
   461→}

   463→/// Ficha de Tarea avanzada — la unidad de conocimiento entrenable.
   464→/// Conecta con la taxonomía vía TaskSubPrinciple ("recetas").
   465→model Task {
   466→  id                 String      @id @default(cuid())
   467→  clubId             String? /// null = biblioteca global de la Cátedra
   468→  createdById        String?
   469→  title              String
   470→  objective          String /// Qué se busca entrenar, en lenguaje de míster
   471→  phases             GamePhase[] /// Fases del juego que trabaja (una tarea puede tocar varias)
   472→  spaceLengthM       Int? /// Dimensiones del espacio: largo (m)
   473→  spaceWidthM        Int? /// Dimensiones del espacio: ancho (m)
   474→  spaceNote          String? /// "Con dos zonas laterales libres"
   475→  playersMin         Int /// Jugadores requeridos (mínimo)
   476→  playersMax         Int? /// Jugadores requeridos (máximo, si aplica)
   477→  durationMin        Int /// Tiempo total sugerido (min)
   478→  repetitions        String? /// "3 series x 6'", "2 bloques x 8'"
   479→  intensity          Intensity   @default(MEDIUM) /// El motor la cruza con la fatiga grupal
   480→  rulesOfProvocation String[] /// Reglas de provocación — array nativo de PostgreSQL
   481→  coachingPoints     String[] /// Puntos clave para el entrenador
   482→  materials          String[] /// Conos, petos, maniquíes...
   483→  diagramUrl         String? /// Diagrama / pizarra de la tarea
   484→  createdAt          DateTime    @default(now())
   485→  updatedAt          DateTime    @updatedAt
   486→
   487→  club          Club?              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   488→  createdBy     User?              @relation("TaskAuthor", fields: [createdById], references: [id], onDelete: SetNull)
   489→  subPrinciples TaskSubPrinciple[] /// Conexiones con la taxonomía ("recetas")
   490→  sessions      SessionTask[] /// Prescripciones históricas en microciclos
   491→
   492→  @@index([clubId])
   493→  @@index([intensity])
   494→}

   496→/// La "receta": conexión tarea ↔ subprincipio con relevancia ponderada.
   497→/// Índice [subPrincipleId, relevance] = LA consulta estrella del motor

   504→  usageNote      String? /// Cuándo usar esta tarea para este subprincipio

   506→
   507→  task         Task         @relation(fields: [taskId], references: [id], onDelete: Cascade)
   508→  subPrinciple SubPrinciple @relation(fields: [subPrincipleId], references: [id], onDelete: Cascade)
   509→
   510→  @@unique([taskId, subPrincipleId])
   511→  @@index([subPrincipleId, relevance])

   518→/// Microciclo semanal: la propuesta cocinada que el míster acepta con un toque.
   519→/// Unique (teamId, weekStart) garantiza que NUNCA haya pantallas en blanco.
   520→model Microcycle {
   521→  id                     String           @id @default(cuid())
   522→  clubId                 String
   523→  teamId                 String
   524→  weekStart              DateTime         @db.Date /// Lunes de la semana ISO
   525→  status                 MicrocycleStatus @default(PROPOSED)
   526→  tacticalDirectives     String[] /// Directrices tácticas de la semana
   527→  methodologicalApproach String? /// Enfoque metodológico global
   528→  diagnosisSummary       String? /// Resumen legible: qué falló + estado del grupo
   529→  engineVersion          String? /// Versión del algoritmo que generó la propuesta
   530→  acceptedById           String?
   531→  acceptedAt             DateTime?
   532→  createdAt              DateTime         @default(now())
   533→  updatedAt              DateTime         @updatedAt
   534→
   535→  club            Club              @relation(fields: [clubId], references: [id], onDelete: Cascade)
   536→  team            Team              @relation(fields: [teamId], references: [id], onDelete: Cascade)
   537→  acceptedBy      User?             @relation("MicrocycleAccepter", fields: [acceptedById], references: [id], onDelete: SetNull)
   538→  analyzedMatches MicrocycleMatch[] /// Partidos cuya ingesta alimentó esta semana
   539→  issues          DiagnosisIssue[] /// Diagnóstico estructurado de la semana
   540→  sessions        Session[] /// Sesiones listas para el césped
   541→
   542→  @@unique([teamId, weekStart])
   543→  @@index([clubId, teamId, weekStart])
   544→}

   546→/// Trazabilidad del motor: qué partidos concretos entraron en cada microciclo

   549→  matchId      String

   551→
   552→  microcycle Microcycle @relation(fields: [microcycleId], references: [id], onDelete: Cascade)
   553→  match      Match      @relation(fields: [matchId], references: [id], onDelete: Cascade)
   554→
   555→  @@id([microcycleId, matchId])
   556→  @@index([matchId])
   557→}

   559→/// Hallazgo del diagnóstico semanal con evidencia.
   560→/// Es el PUENTE entre los datos crudos (Pilar 3) y la prescripción (tareas).
   561→model DiagnosisIssue {
   562→  id             String      @id @default(cuid())
   563→  microcycleId   String
   564→  subPrincipleId String? /// Qué subprincipio falla (null = hallazgo físico/anímico)
   565→  phase          GamePhase? /// Fase donde se detecta el problema
   566→  kind           IssueKind
   567→  severity       Int         @default(2) /// 1-5 (5 = urgente) → prioriza la prescripción
   568→  evidence       String /// Evidencia legible: "9 pérdidas en salida por zona izquierda"
   569→  metric         Json? /// Números crudos del agregado (auditoría del motor)
   570→  status         IssueStatus @default(OPEN)
   571→  createdAt      DateTime    @default(now())
   572→
   573→  microcycle   Microcycle    @relation(fields: [microcycleId], references: [id], onDelete: Cascade)
   574→  subPrinciple SubPrinciple? @relation(fields: [subPrincipleId], references: [id], onDelete: SetNull)
   575→  prescribedIn SessionTask[] /// Tareas que lo abordan (trazabilidad receta → césped)
   576→
   577→  @@index([microcycleId])
   578→  @@index([subPrincipleId])
   579→}

   581→/// Sesión de entrenamiento dentro del microciclo (calendario MD+1, MD-3...)
   582→model Session {
   583→  id           String   @id @default(cuid())
   584→  microcycleId String
   585→  date         DateTime @db.Date
   586→  label        String /// "MD+1", "MD-3", "MD-1" — código del calendario semanal
   587→  objective    String? /// Objetivo de la sesión en una frase
   588→  durationMin  Int?
   589→  createdAt    DateTime @default(now())
   590→
   591→  microcycle Microcycle    @relation(fields: [microcycleId], references: [id], onDelete: Cascade)
   592→  tasks      SessionTask[] /// Tareas exactas listas para el césped
   593→
   594→  @@index([microcycleId, date])
   595→}
   596→
   597→/// Tarea prescrita en una sesión — el punto final de la cadena del motor.
   598→/// `prescribedIssueId` da trazabilidad total: por qué esta tarea está aquí.
   599→model SessionTask {
   600→  id                String   @id @default(cuid())
   601→  sessionId         String
   602→  taskId            String
   603→  order             Int /// Orden dentro de la sesión (calentamiento, principal, cierre)
   604→  durationMin       Int? /// Tiempo real asignado a la tarea
   605→  rationale         String? /// "Por qué esta tarea": texto que el motor genera para el míster
   606→  prescribedIssueId String? /// Hallazgo que esta tarea ayuda a corregir
   607→  createdAt         DateTime @default(now())
   608→
   609→  session Session         @relation(fields: [sessionId], references: [id], onDelete: Cascade)
   610→  task    Task            @relation(fields: [taskId], references: [id], onDelete: Cascade)
   611→  issue   DiagnosisIssue? @relation(fields: [prescribedIssueId], references: [id], onDelete: SetNull)
   612→
   613→  @@unique([sessionId, order])
   614→  @@index([taskId])
   615→}

   617→/// Auditoría de cada ejecución del motor (robustez interna, invisible al usuario)
   618→model EngineRun {

   621→  teamId        String

   629→  durationMs    Int? /// Tiempo de cómputo
   630→  createdAt     DateTime @default(now())
   631→
   632→  club Club @relation(fields: [clubId], references: [id], onDelete: Cascade)
   633→  team Team @relation(fields: [teamId], references: [id], onDelete: Cascade)
   634→
   635→  @@index([teamId, createdAt])
   636→  @@index([microcycleId])

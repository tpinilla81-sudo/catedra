     2→// MOTOR PRESCRIPTIVO v0.2 — Pipeline completo: diagnóstico → receta → plan
     3→// ----------------------------------------------------------------------------
     4→// Cierra el Pilar 4 end-to-end. Sobre el diagnóstico heurístico de
     5→// `diagnose.ts`, este módulo añade:
     6→//   1. RANKING DE CANDIDATAS: por cada hallazgo evalúa TODAS las tareas
     7→//      conectadas en la receta (TaskSubPrinciple), ordenadas por relevancia.
     8→//   2. FACTIBILIDAD: descarta tareas inviables (jugadores disponibles
     9→//      insuficientes, intensidad alta contraindicada por fatiga grupal).
    10→//      Aplica TAMBIÉN a las sesiones de apoyo MD+1 (recuperación, solo LOW)
    11→//      y MD-1 (activación, LOW/MEDIUM con techo por fatiga) — ninguna tarea
    12→//      llega al césped sin cumplir los requisitos físicos del grupo.
    13→//   3. AUDITORÍA TOTAL: cada ejecución persiste un EngineRun con las entradas,
    14→//      los hallazgos, las tareas elegidas (con rationale) y las descartadas
    15→//      (con motivo) — trazabilidad "por qué este plan" de punta a punta.
    16→//   4. RE-PRESCRIPCIÓN EN VIVO: cuando llega evidencia nueva (ingesta) a un
    17→//      microciclo aún PROPOSED, regenera diagnóstico Y tareas de las sesiones
    18→//      principales (MD-3 / MD-2), respetando los planes ya aceptados.
    19→//
    20→// Consumidores:
    21→//   • today/actions.ts   → ensureWeeklyProposal (vista Hoy, lunes)
    22→//   • match/actions.ts   → submitMatchReport (hook de ingesta)
    23→//   • scripts/validate-pilar4.ts (validación E2E)

    88→const INTENSITY_ORDER: Record<string, number> = { LOW: 0, MEDIUM: 1, HIGH: 2 };

    90→// ============================================================================
    91→// 1) RANKING DE CANDIDATAS (receta ponderada de la Biblioteca)
    92→// ============================================================================
    93→
    94→/**
    95→ * Devuelve todas las tareas conectadas a un subprincipio por la receta,
    96→ * ordenadas por: relevancia (3 "Núcleo" primero) → intensidad suave antes que
    97→ * alta → título (determinismo).
    98→ */
    99→export async function rankTaskCandidates(
   100→  db: PipelineDb,
   101→  subPrincipleId: string,
   102→): Promise<
   103→  Array<{
   104→    taskId: string;
   105→    taskTitle: string;
   106→    relevance: number;
   107→    intensity: string;
   108→    playersMin: number;
   109→    playersMax: number | null;
   110→    durationMin: number;
   111→    usageNote: string | null;
   112→  }>
   113→> {
   114→  const links = await db.taskSubPrinciple.findMany({
   115→    where: { subPrincipleId },
   116→    include: { task: { select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true } } },
   117→  });
   118→  return links
   119→    .map((l) => ({
   120→      taskId: l.task.id,
   121→      taskTitle: l.task.title,
   122→      relevance: l.relevance,
   123→      intensity: l.task.intensity as string,
   124→      playersMin: l.task.playersMin,
   125→      playersMax: l.task.playersMax,
   126→      durationMin: l.task.durationMin,
   127→      usageNote: l.usageNote,
   128→    }))
   129→    .sort(
   130→      (a, b) =>
   131→        b.relevance - a.relevance ||
   132→        (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   133→        a.taskTitle.localeCompare(b.taskTitle),
   134→    );
   135→}

   137→// ============================================================================
   138→// 2) PRESCRIPCIÓN CON FACTIBILIDAD — elegida + descartadas con motivo
   139→// ============================================================================
   140→
   141→type CandidateRow = {
   142→  taskId: string;
   143→  taskTitle: string;
   144→  relevance: number;
   145→  intensity: string;
   146→  playersMin: number;
   147→  playersMax: number | null;
   148→  durationMin: number;
   149→  usageNote: string | null;
   150→};
   151→
   152→/**
   153→ * Recorre las candidatas en orden de ranking y devuelve el índice de la
   154→ * primera FACTIBLE. Las inviables se auditan en `sink` (sin duplicados):
   155→ *   • fatiga ≥ 7 + intensidad HIGH → contraindicada (cruce con el Pilar 2)
   156→ *   • playersMin > disponibles     → inviable con la plantilla de hoy
   157→ */
   158→function feasibilityScan(
   159→  candidates: CandidateRow[],
   160→  opts: { availablePlayers: number; fatigueScore: number },
   161→  sink: DiscardedCandidate[],
   162→  seen: Set<string>,
   163→): number {
   164→  for (let c = 0; c < candidates.length; c++) {
   165→    const cand = candidates[c];
   166→    let reason: string | null = null;
   167→    if (opts.fatigueScore >= 7 && cand.intensity === "HIGH") {
   168→      reason = `Intensidad alta contraindicada con fatiga grupal ${opts.fatigueScore}/10`;
   169→    } else if (cand.playersMin > opts.availablePlayers) {
   170→      reason = `Necesita ${cand.playersMin} jugadores y solo hay ${opts.availablePlayers} disponibles`;
   171→    }
   172→    if (!reason) return c;
   173→    const key = `${cand.taskId}|${reason}`;
   174→    if (!seen.has(key)) {
   175→      seen.add(key);
   176→      sink.push({
   177→        taskId: cand.taskId,
   178→        taskTitle: cand.taskTitle,
   179→        relevance: cand.relevance,
   180→        reason,
   181→      });
   182→    }
   183→  }
   184→  return -1;
   185→}
   186→
   187→/** Orden canónico de la receta: relevancia → intensidad → título */
   188→function sortCandidates(list: CandidateRow[]): CandidateRow[] {
   189→  return list.sort(
   190→    (a, b) =>
   191→      b.relevance - a.relevance ||
   192→      (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   193→      a.taskTitle.localeCompare(b.taskTitle),
   194→  );
   195→}
   196→
   197→/**
   198→ * Candidatas de TODO el pool de la fase (todos los subprincipios visibles de
   199→ * esa fase, sin duplicar tareas). Es la red de seguridad del fallback: si el
   200→ * subprincipio exacto no tiene tarea factible, la fase sí puede tenerla.
   201→ */
   202→async function rankPhaseCandidates(
   203→  db: PipelineDb,
   204→  phase: string,
   205→): Promise<CandidateRow[]> {
   206→  const sps = await db.subPrinciple.findMany({
   207→    where: { phase: phase as GamePhase },
   208→    select: { id: true },
   209→    // Determinismo: mismo pool de subprincipios en cada ejecución
   210→    orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   211→  });
   212→  if (sps.length === 0) return [];
   213→  const links = await db.taskSubPrinciple.findMany({
   214→    where: { subPrincipleId: { in: sps.map((s) => s.id) } },
   215→    // Determinismo del dedupe por tarea (a igual relevancia gana el primero)
   216→    orderBy: [{ relevance: "desc" }, { id: "asc" }],
   217→    include: {
   218→      task: {
   219→        select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true },
   220→      },
   221→    },
   222→  });
   223→  const byTask = new Map<string, CandidateRow>();
   224→  for (const l of links) {
   225→    const prev = byTask.get(l.task.id);
   226→    if (!prev || l.relevance > prev.relevance) {
   227→      byTask.set(l.task.id, {
   228→        taskId: l.task.id,
   229→        taskTitle: l.task.title,
   230→        relevance: l.relevance,
   231→        intensity: l.task.intensity as string,
   232→        playersMin: l.task.playersMin,
   233→        playersMax: l.task.playersMax,
   234→        durationMin: l.task.durationMin,
   235→        usageNote: l.usageNote,
   236→      });
   237→    }
   238→  }
   239→  return sortCandidates([...byTask.values()]);
   240→}
   241→
   242→/**
   243→ * Por cada hallazgo con subprincipio asociado, recorre la receta en orden de
   244→ * ranking y elige la primera tarea FACTIBLE. Si el subprincipio exacto se
   245→ * queda sin candidatas viables, amplía a la mejor tarea de la FASE completa
   246→ * (receta ampliada, auditable). El resto se audita como descartadas.
   247→ */
   248→export async function prescribeForIssues(
   249→  db: PipelineDb,
   250→  drafts: IssueDraft[],
   251→  opts: { availablePlayers: number; fatigueScore: number },
   252→): Promise<{
   253→  prescriptions: Array<{
   254→    issueIndex: number;
   255→    taskId: string;
   256→    taskTitle: string;
   257→    relevance: number;
   258→    usageNote: string | null;
   259→    durationMin: number;
   260→    rationale: string;
   261→  }>;
   262→  discarded: DiscardedCandidate[];
   263→}> {
   264→  const prescriptions: Array<{
   265→    issueIndex: number;
   266→    taskId: string;
   267→    taskTitle: string;
   268→    relevance: number;
   269→    usageNote: string | null;
   270→    durationMin: number;
   271→    rationale: string;
   272→  }> = [];
   273→  const discarded: DiscardedCandidate[] = [];
   274→  const seen = new Set<string>();
   275→  const { availablePlayers, fatigueScore } = opts;
   276→
   277→  for (let i = 0; i < drafts.length; i++) {
   278→    const draft = drafts[i];
   279→    if (!draft.subPrincipleId) continue;
   280→
   281→    let candidates = await rankTaskCandidates(db, draft.subPrincipleId);
   282→    let chosenIndex = feasibilityScan(candidates, opts, discarded, seen);
   283→    let fallbackNote = "";
   284→
   285→    // Fallback por fase: ningún subprincipio exacto tenía candidata factible
   286→    if (chosenIndex === -1 && draft.phase) {
   287→      const phaseCandidates = await rankPhaseCandidates(db, draft.phase);
   288→      chosenIndex = feasibilityScan(phaseCandidates, opts, discarded, seen);
   289→      if (chosenIndex !== -1) {
   290→        candidates = phaseCandidates;
   291→        fallbackNote = " Receta ampliada a la fase: el subprincipio exacto no tenía tarea factible.";
   292→      }
   293→    }
   294→
   295→    if (chosenIndex === -1) continue;
   296→
   297→    const chosen = candidates[chosenIndex];
   298→    // El resto de candidatas (detrás de la elegida) quedan auditadas
   299→    for (let c = chosenIndex + 1; c < candidates.length; c++) {
   300→      const cand = candidates[c];
   301→      const reason =
   302→        cand.relevance < chosen.relevance
   303→          ? `Relevancia menor en la receta (${cand.relevance} vs ${chosen.relevance})`
   304→          : "Priorizada otra candidata de mayor afinidad";
   305→      const key = `${cand.taskId}|${reason}`;
   306→      if (!seen.has(key)) {
   307→        seen.add(key);
   308→        discarded.push({
   309→          taskId: cand.taskId,
   310→          taskTitle: cand.taskTitle,
   311→          relevance: cand.relevance,
   312→          reason,
   313→        });
   314→      }
   315→    }
   316→
   317→    const intensityFit =
   318→      fatigueScore >= 7
   319→        ? "Intensidad ajustada a la fatiga del grupo."
   320→        : "";
   321→    prescriptions.push({
   322→      issueIndex: i,
   323→      taskId: chosen.taskId,
   324→      taskTitle: chosen.taskTitle,
   325→      relevance: chosen.relevance,
   326→      usageNote: chosen.usageNote,
   327→      durationMin: Math.min(chosen.durationMin, draft.kind === "PHYSICAL" ? 40 : 60),
   328→      rationale: `Entra por: ${draft.evidence}. ${chosen.usageNote ?? ""} ${intensityFit}${fallbackNote}`.trim(),
   329→    });
   330→  }
   331→
   332→  return { prescriptions, discarded };
   333→}

   381→export type ProposalResult = {
   382→  created: boolean;
   383→  microcycleId: string | null;
   384→  runId: string | null;
   385→  issuesCount: number;
   386→  prescriptions: PrescriptionAudit[];
   387→  discarded: DiscardedCandidate[];
   388→  durationMs: number | null;
   389→};
   390→
   391→/**
   392→ * Cocina el microciclo PROPOSED de `weekStart` desde los datos reales y
   393→ * persiste la auditoría completa en EngineRun. Si ya existe un microciclo
   394→ * para esa semana devuelve `created: false` sin tocar nada.
   395→ *
   396→ * TODO el pipeline corre en UNA transacción interactiva: si otro request
   397→ * concurrente cocina la misma semana (carrera TOCTOU), la constraint única
   398→ * (teamId, weekStart) lanza P2002 → la transacción perdedora se revierte
   399→ * entera y se devuelve `created: false` sin error 500 al usuario.
   400→ */
   401→export async function runWeeklyProposalPipeline(
   402→  db: PipelineRunnerDb,
   403→  args: { clubId: string; teamId: string; weekStart: Date },
   404→): Promise<ProposalResult> {
   405→  const t0 = Date.now();
   406→  const { clubId, teamId, weekStart } = args;
   407→
   408→  try {
   409→    return await db.$transaction(async (tx) => {
   410→      const existing = await tx.microcycle.findUnique({
   411→        where: { teamId_weekStart: { teamId, weekStart } },
   412→        select: { id: true },
   413→      });
   414→      if (existing) {
   415→        return {
   416→          created: false,
   417→          microcycleId: existing.id,
   418→          runId: null,
   419→          issuesCount: 0,
   420→          prescriptions: [],
   421→          discarded: [],
   422→          durationMs: null,
   423→        };
   424→      }
   425→
   426→      // ---------- 1) Entradas ----------
   427→      const inputs = await collectInputs(tx, teamId, weekStart);
   428→      const pulse = inputs.pulse;
   429→      const recentMatches = inputs.matches;
   430→
   431→      const availablePlayers = await tx.player.count({
   432→        where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   433→      });
   434→
   435→      // ---------- 2) Diagnóstico (heurística compartida) ----------
   436→      const drafts = await buildIssueDrafts(tx, teamId, inputs);
   437→      const last = recentMatches[0];
   438→
   439→      // ---------- 3) Prescripción con ranking + factibilidad ----------
   440→      const { prescriptions: rawRx, discarded } = await prescribeForIssues(tx, drafts, {
   441→        availablePlayers,
   442→        fatigueScore: pulse?.fatigueScore ?? 5,
   443→      });
   444→
   445→      const auditRx: PrescriptionAudit[] = rawRx.map((p) => ({
   446→        issueEvidence: drafts[p.issueIndex].evidence,
   447→        issueKind: drafts[p.issueIndex].kind,
   448→        subPrincipleId: drafts[p.issueIndex].subPrincipleId ?? null,
   449→        taskId: p.taskId,
   450→        taskTitle: p.taskTitle,
   451→        relevance: p.relevance,
   452→        sessionLabel: "",
   453→        durationMin: p.durationMin,
   454→        rationale: p.rationale,
   455→      }));
   456→
   457→      // ---------- 4) Tareas de apoyo (MD+1 / MD-1) CON FACTIBILIDAD FÍSICA ----------
   458→      // La factibilidad del grupo (jugadores disponibles + umbrales de fatiga)
   459→      // NO es exclusiva de la prescripción por hallazgo: las sesiones de
   460→      // recuperación y activación también respetan los requisitos físicos.
   461→      //   • MD+1 recuperación: SOLO intensidad LOW (regeneración) y
   462→      //     playersMin ≤ disponibles.
   463→      //   • MD-1 activación: LOW/MEDIUM, pero con fatiga grupal ≥ 7 se
   464→      //     restringe a LOW (no añadir carga pre-partido) y playersMin ≤
   465→      //     disponibles. Sin candidata factible → sesión sin tarea (la UI la
   466→      //     cubre como "sesión libre"): nunca una tarea inviable en el césped.
   467→      type SupportTask = { id: string; title: string; intensity: string; playersMin: number };
   468→      const supportSelect = { id: true, title: true, intensity: true, playersMin: true } as const;
   469→      const fatigueHigh = (pulse?.fatigueScore ?? 5) >= 7;
   470→      const recoveryPool: SupportTask[] = await tx.task.findMany({
   471→        where: { intensity: "LOW", phases: { has: "OFFENSIVE" } },
   472→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   473→        select: supportSelect,
   474→      });
   475→      const activationPool: SupportTask[] = await tx.task.findMany({
   476→        where: { intensity: { in: ["LOW", "MEDIUM"] }, phases: { has: "OFFENSIVE" } },
   477→        orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   478→        select: supportSelect,
   479→      });
   480→      const firstFeasibleSupport = (pool: SupportTask[], allowMedium: boolean): SupportTask | null =>
   481→        pool.find(
   482→          (t) =>
   483→            t.playersMin <= availablePlayers && // plantilla suficiente hoy
   484→            (!fatigueHigh || t.intensity === "LOW") && // techo de intensidad por fatiga
   485→            (allowMedium || t.intensity === "LOW"), // techo de intensidad de la sesión
   486→        ) ?? null;
   487→      const recoveryTask = firstFeasibleSupport(recoveryPool, false);
   488→      const activationTask = firstFeasibleSupport(activationPool, true);
   489→
   490→      // ---------- 5) Construcción del microciclo ----------
   491→      const tacticalDrafts = drafts.filter((d) => d.kind === "TACTICAL");
   492→      const day = (offset: number) => {
   493→        const d = new Date(weekStart);
   494→        d.setDate(d.getDate() + offset);
   495→        return d;
   496→      };
   497→
   498→      const directives = tacticalDrafts.length
   499→        ? tacticalDrafts.map((d) => `Corregir: ${d.evidence}`)
   500→        : ["Consolidar la identidad de juego de la semana"];
   501→
   502→      const summary = [
   503→        `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   504→        pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   505→        `${availablePlayers} jugadores disponibles.`,
   506→        `Prioridad de la semana: ${directives[0]}.`,
   507→      ]
   508→        .filter(Boolean)
   509→        .join(" ");
   510→
   511→      const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   512→
   513→      const microcycle = await tx.microcycle.create({
   514→        data: {
   515→          clubId,
   516→          teamId,
   517→          weekStart,
   518→          status: "PROPOSED",
   519→          tacticalDirectives: directives,
   520→          methodologicalApproach: "Analítico-inductivo por proximidad, progresión a tareas globales",
   521→          diagnosisSummary: summary,
   522→          engineVersion: ENGINE_VERSION,
   523→          analyzedMatches: {
   524→            create: recentMatches.map((m) => ({ matchId: m.id })),
   525→          },
   526→          issues: {
   527→            create: drafts.map((d) => ({
   528→              subPrincipleId: d.subPrincipleId,
   529→              phase: d.phase,
   530→              kind: d.kind,
   531→              severity: d.severity,
   532→              evidence: d.evidence,
   533→              metric: d.metric as never,
   534→            })),
   535→          },
   536→          sessions: {
   537→            create: [
   538→              // MD+1 — Regeneración
   539→              {
   540→                date: day(1),
   541→                label: "MD+1",
   542→                objective: "Regeneración y asimilación del esfuerzo del partido",
   543→                durationMin: 50,
   544→                tasks: {
   545→                  create: recoveryTask
   546→                    ? [
   547→                        {
   548→                          order: 1,
   549→                          taskId: recoveryTask.id,
   550→                          durationMin: 40,
   551→                          rationale:
   552→                            (pulse?.fatigueScore ?? 5) >= 7
   553→                              ? "Fatiga alta: intensidad controlada para recuperar al grupo."
   554→                              : "Volumen bajo para asimilar el partido y mantener toque de balón.",
   555→                        },
   556→                      ]
   557→                    : [],
   558→                },
   559→              },
   560→              // MD-3 — Sesión principal (táctica)
   561→              {
   562→                date: day(2),
   563→                label: "MD-3",
   564→                objective: tacticalDrafts[0]?.evidence ?? "Trabajo táctico principal",
   565→                durationMin: 80,
   566→                tasks: {
   567→                  create: tacticalRx.slice(0, 2).map((p, idx) => ({
   568→                    order: idx + 1,
   569→                    taskId: p.taskId,
   570→                    durationMin: p.durationMin,
   571→                    rationale: p.rationale,
   572→                  })),
   573→                },
   574→              },
   575→              // MD-2 — Consolidación
   576→              {
   577→                date: day(3),
   578→                label: "MD-2",
   579→                objective: "Consolidación + preparación del plan de partido",
   580→                durationMin: 70,
   581→                tasks: {
   582→                  create: tacticalRx.slice(1, 2).map((p) => ({
   583→                    order: 1,
   584→                    taskId: p.taskId,
   585→                    durationMin: p.durationMin,
   586→                    rationale: p.rationale,
   587→                  })),
   588→                },
   589→              },
   590→              // MD-1 — Activación pre-partido
   591→              {
   592→                date: day(4),
   593→                label: "MD-1",
   594→                objective: "Activación pre-partido: velocidad y automatismos",
   595→                durationMin: 45,
   596→                tasks: {
   597→                  create: activationTask
   598→                    ? [
   599→                        {
   600→                          order: 1,
   601→                          taskId: activationTask.id,
   602→                          durationMin: 35,
   603→                          rationale: fatigueHigh
   604→                            ? `Activación neuromuscular previa al partido: intensidad baja exigida por la fatiga grupal (${pulse?.fatigueScore}/10) y plantilla disponible (${availablePlayers}).`
   605→                            : "Activación neuromuscular previa al partido, factible con la plantilla disponible.",
   606→                        },
   607→                      ]
   608→                    : [],
   609→                },
   610→              },
   611→            ],
   612→          },
   613→        },
   614→        include: { issues: true, sessions: { include: { tasks: true } } },
   615→      });
   616→
   617→      // ---------- 6) Trazabilidad hallazgo → tarea (prescribedIssueId) ----------
   618→      // La primera prescripción táctica entra en MD-3; la segunda aparece en MD-3
   619→      // (profundizar) Y en MD-2 (consolidar). Se enlaza el hallazgo en AMBAS.
   620→      const evidenceToIssue = new Map(microcycle.issues.map((iss) => [iss.evidence, iss.id]));
   621→      const md3 = microcycle.sessions.find((s) => s.label === "MD-3");
   622→      const md2 = microcycle.sessions.find((s) => s.label === "MD-2");
   623→      const sessionByLabel = { "MD-3": md3, "MD-2": md2 } as Record<string, typeof md3>;
   624→
   625→      for (const label of ["MD-3", "MD-2"] as const) {
   626→        const session = sessionByLabel[label];
   627→        if (!session) continue;
   628→        const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   629→        for (const p of forSession) {
   630→          const draft = drafts[p.issueIndex];
   631→          const issueId = evidenceToIssue.get(draft.evidence);
   632→          const st = session.tasks.find(
   633→            (t) => t.taskId === p.taskId && t.rationale === p.rationale,
   634→          );
   635→          if (!st) continue;
   636→          await tx.sessionTask.update({
   637→            where: { id: st.id },
   638→            data: { prescribedIssueId: issueId },
   639→          });
   640→          const audit = auditRx.find(
   641→            (a) => a.taskId === p.taskId && a.issueEvidence === draft.evidence,
   642→          );
   643→          if (audit && !audit.sessionLabel) audit.sessionLabel = label;
   644→        }
   645→      }
   646→
   647→      // ---------- 7) Auditoría de la ejecución ----------
   648→      const inputSnapshot = {
   649→        weekStart: weekStart.toISOString().slice(0, 10),
   650→        pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   651→        matches: recentMatches.map((m) => ({
   652→          opponent: m.opponent,
   653→          score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   654→          events: m.report?.events.length ?? 0,
   655→        })),
   656→        availablePlayers,
   657→      };
   658→      const runId = await recordEngineRun(tx, {
   659→        clubId,
   660→        teamId,
   661→        microcycleId: microcycle.id,
   662→        trigger: "WEEKLY_PROPOSAL",
   663→        t0,
   664→        inputs: inputSnapshot,
   665→        drafts,
   666→        prescriptions: auditRx,
   667→        discarded,
   668→      });
   669→
   670→      return {
   671→        created: true,
   672→        microcycleId: microcycle.id,
   673→        runId,
   674→        issuesCount: drafts.length,
   675→        prescriptions: auditRx,
   676→        discarded,
   677→        durationMs: Date.now() - t0,
   678→      };
   679→    }, TX_OPTS);
   680→  } catch (e) {
   681→    // Carrera TOCTOU: otro request cocinó la misma semana a la vez. La tx
   682→    // perdedora se revierte ENTERA (sin microciclo ni EngineRun parcial) y se
   683→    // responde created:false — nunca un 500 sin capturar al usuario.
   684→    if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   685→      const existing = await db.microcycle.findUnique({
   686→        where: { teamId_weekStart: { teamId, weekStart } },
   687→        select: { id: true },
   688→      });
   689→      return {
   690→        created: false,
   691→        microcycleId: existing?.id ?? null,
   692→        runId: null,
   693→        issuesCount: 0,
   694→        prescriptions: [],
   695→        discarded: [],
   696→        durationMs: null,
   697→      };
   698→    }
   699→    throw e;
   700→  }
   701→}
   702→
   703→// ============================================================================
   704→// 5) RE-PRESCRIPCIÓN EN VIVO — evidencia nueva → propuesta PROPOSED actualizada
   705→// ============================================================================
   706→
   707→export type RefreshResult = {
   708→  refreshedWeeks: string[];
   709→  evidences: string[];
   710→  runs: number;
   711→};
   712→
   713→/**
   714→ * Hook de conexión (Pilar 3 → Pilar 4). Se ejecuta al guardar una ingesta.
   715→ * Para cada microciclo aún PROPOSED de la semana del partido o la siguiente:
   716→ *   1. Regenera el diagnóstico (issues + directrices + resumen).
   717→ *   2. RE-PRESCRIBE las tareas de las sesiones principales (MD-3 / MD-2)
   718→ *      con la receta fresca y la factibilidad de hoy.
   719→ *   3. Persiste un EngineRun (trigger EVIDENCE_REFRESH) con toda la auditoría.
   720→ * Los planes ACCEPTED/ADJUSTED se respetan: nunca se tocan.
   721→ *
   722→ * Cada semana se procesa en SU propia transacción: diagnóstico, tareas y
   723→ * auditoría de esa semana se escriben todos o ninguno.
   724→ */
   725→export async function refreshProposalEvidence(
   726→  db: PipelineRunnerDb,
   727→  args: { clubId: string; teamId: string; kickoffAt: Date },
   728→): Promise<RefreshResult> {
   729→  const { clubId, teamId, kickoffAt } = args;
   730→  const base = mondayOf(kickoffAt);
   731→  const candidates = [base, addDays(base, 7)]; // misma semana y semana siguiente
   732→  const refreshedWeeks: string[] = [];
   733→  const evidences: string[] = [];
   734→  let runs = 0;
   735→
   736→  for (const weekStart of candidates) {
   737→    const touched = await db.$transaction(async (tx) => {
   738→      const t0 = Date.now();
   739→      const cycle = await tx.microcycle.findUnique({
   740→        where: { teamId_weekStart: { teamId, weekStart } },
   741→        include: { sessions: { include: { tasks: true } } },
   742→      });
   743→      // Solo tocamos propuestas NO validadas por el cuerpo técnico
   744→      if (!cycle || cycle.status !== "PROPOSED") return false;
   745→
   746→      const inputs = await collectInputs(tx, teamId, weekStart);
   747→      const pulse = inputs.pulse;
   748→      const drafts = await buildIssueDrafts(tx, teamId, inputs);
   749→
   750→      const availablePlayers = await tx.player.count({
   751→        where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   752→      });
   753→
   754→      // 1) Diagnóstico regenerado
   755→      await tx.diagnosisIssue.deleteMany({
   756→        where: { microcycleId: cycle.id, status: "OPEN" },
   757→      });
   758→      await tx.diagnosisIssue.createMany({
   759→        data: drafts.map((d) => ({
   760→          microcycleId: cycle.id,
   761→          subPrincipleId: d.subPrincipleId,
   762→          phase: d.phase,
   763→          kind: d.kind,
   764→          severity: d.severity,
   765→          evidence: d.evidence,
   766→          metric: d.metric as never,
   767→        })),
   768→      });
   769→
   770→      // 2) Re-prescripción con ranking + factibilidad
   771→      const { prescriptions: rawRx, discarded } = await prescribeForIssues(tx, drafts, {
   772→        availablePlayers,
   773→        fatigueScore: pulse?.fatigueScore ?? 5,
   774→      });
   775→      const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   776→
   777→      // Reemplaza las tareas de MD-3 / MD-2 con la prescripción fresca
   778→      const issueRows = await tx.diagnosisIssue.findMany({
   779→        where: { microcycleId: cycle.id, status: "OPEN" },
   780→        select: { id: true, evidence: true },
   781→        // Determinismo: mapeo estable evidencia → hallazgo
   782→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   783→      });
   784→      const evidenceToIssue = new Map(issueRows.map((r) => [r.evidence, r.id]));
   785→      const auditRx: PrescriptionAudit[] = [];
   786→
   787→      for (const label of ["MD-3", "MD-2"] as const) {
   788→        const session = cycle.sessions.find((s) => s.label === label);
   789→        if (!session) continue;
   790→        await tx.sessionTask.deleteMany({ where: { sessionId: session.id } });
   791→
   792→        const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   793→        for (let idx = 0; idx < forSession.length; idx++) {
   794→          const p = forSession[idx];
   795→          const draft = drafts[p.issueIndex];
   796→          const issueId = evidenceToIssue.get(draft.evidence) ?? null;
   797→          await tx.sessionTask.create({
   798→            data: {
   799→              sessionId: session.id,
   800→              taskId: p.taskId,
   801→              order: idx + 1,
   802→              durationMin: p.durationMin,
   803→              rationale: p.rationale,
   804→              prescribedIssueId: issueId,
   805→            },
   806→          });
   807→          auditRx.push({
   808→            issueEvidence: draft.evidence,
   809→            issueKind: draft.kind,
   810→            subPrincipleId: draft.subPrincipleId ?? null,
   811→            taskId: p.taskId,
   812→            taskTitle: p.taskTitle,
   813→            relevance: p.relevance,
   814→            sessionLabel: label,
   815→            durationMin: p.durationMin,
   816→            rationale: p.rationale,
   817→          });
   818→        }
   819→      }
   820→
   821→      // 3) Directrices + resumen legible
   822→      const last = inputs.matches[0];
   823→      const directives = tacticalRx.length
   824→        ? tacticalRx.map((p) => `Corregir: ${drafts[p.issueIndex].evidence}`)
   825→        : ["Consolidar la identidad de juego de la semana"];
   826→      await tx.microcycle.update({
   827→        where: { id: cycle.id },
   828→        data: {
   829→          tacticalDirectives: directives,
   830→          diagnosisSummary: [
   831→            `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   832→            pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   833→            `Prioridad de la semana: ${directives[0]}.`,
   834→          ]
   835→            .filter(Boolean)
   836→            .join(" "),
   837→          engineVersion: ENGINE_VERSION,
   838→        },
   839→      });
   840→
   841→      // 4) Auditoría EVIDENCE_REFRESH
   842→      await recordEngineRun(tx, {
   843→        clubId,
   844→        teamId,
   845→        microcycleId: cycle.id,
   846→        trigger: "EVIDENCE_REFRESH",
   847→        t0,
   848→        inputs: {
   849→          weekStart: weekStart.toISOString().slice(0, 10),
   850→          pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   851→          matches: inputs.matches.map((m) => ({
   852→            opponent: m.opponent,
   853→            score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   854→            events: m.report?.events.length ?? 0,
   855→          })),
   856→          availablePlayers,
   857→        },
   858→        drafts,
   859→        prescriptions: auditRx,
   860→        discarded,
   861→      });
   862→
   863→      refreshedWeeks.push(weekStart.toISOString().slice(0, 10));
   864→      for (const d of drafts.slice(0, 3)) evidences.push(d.evidence);
   865→      return true;
   866→    }, TX_OPTS);
   867→
   868→    if (touched) runs++;
   869→  }
   870→
   871→  return { refreshedWeeks, evidences, runs };
   872→}

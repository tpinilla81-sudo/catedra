     4→// Cierra el Pilar 4 end-to-end. Sobre el diagnóstico heurístico de
     5→// `diagnose.ts`, este módulo añade:

     7→//      conectadas en la receta (TaskSubPrinciple), ordenadas por relevancia.
     8→//   2. FACTIBILIDAD: descarta tareas inviables (jugadores disponibles
     9→//      insuficientes, intensidad alta contraindicada por fatiga grupal).
    10→//   3. AUDITORÍA TOTAL: cada ejecución persiste un EngineRun con las entradas,
    11→//      los hallazgos, las tareas elegidas (con rationale) y las descartadas
    12→//      (con motivo) — trazabilidad "por qué este plan" de punta a punta.
    13→//   4. RE-PRESCRIPCIÓN EN VIVO: cuando llega evidencia nueva (ingesta) a un
    14→//      microciclo aún PROPOSED, regenera diagnóstico Y tareas de las sesiones
    15→//      principales (MD-3 / MD-2), respetando los planes ya aceptados.
    16→//
    17→// Consumidores:
    18→//   • today/actions.ts   → ensureWeeklyProposal (vista Hoy, lunes)
    19→//   • match/actions.ts   → submitMatchReport (hook de ingesta)
    20→//   • scripts/validate-pilar4.ts (validación E2E)

    72→const INTENSITY_ORDER: Record<string, number> = { LOW: 0, MEDIUM: 1, HIGH: 2 };

    74→// ============================================================================
    75→// 1) RANKING DE CANDIDATAS (receta ponderada de la Biblioteca)
    76→// ============================================================================
    77→
    78→/**
    79→ * Devuelve todas las tareas conectadas a un subprincipio por la receta,
    80→ * ordenadas por: relevancia (3 "Núcleo" primero) → intensidad suave antes que
    81→ * alta → título (determinismo).
    82→ */
    83→export async function rankTaskCandidates(
    84→  db: PipelineDb,
    85→  subPrincipleId: string,
    86→): Promise<
    87→  Array<{
    88→    taskId: string;
    89→    taskTitle: string;
    90→    relevance: number;
    91→    intensity: string;
    92→    playersMin: number;
    93→    playersMax: number | null;
    94→    durationMin: number;
    95→    usageNote: string | null;
    96→  }>
    97→> {
    98→  const links = await db.taskSubPrinciple.findMany({
    99→    where: { subPrincipleId },
   100→    include: { task: { select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true } } },
   101→  });
   102→  return links
   103→    .map((l) => ({
   104→      taskId: l.task.id,
   105→      taskTitle: l.task.title,
   106→      relevance: l.relevance,
   107→      intensity: l.task.intensity as string,
   108→      playersMin: l.task.playersMin,
   109→      playersMax: l.task.playersMax,
   110→      durationMin: l.task.durationMin,
   111→      usageNote: l.usageNote,
   112→    }))
   113→    .sort(
   114→      (a, b) =>
   115→        b.relevance - a.relevance ||
   116→        (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   117→        a.taskTitle.localeCompare(b.taskTitle),
   118→    );
   119→}
   120→
   121→// ============================================================================
   122→// 2) PRESCRIPCIÓN CON FACTIBILIDAD — elegida + descartadas con motivo
   123→// ============================================================================
   124→
   125→type CandidateRow = {
   126→  taskId: string;
   127→  taskTitle: string;
   128→  relevance: number;
   129→  intensity: string;
   130→  playersMin: number;
   131→  playersMax: number | null;
   132→  durationMin: number;
   133→  usageNote: string | null;
   134→};
   135→
   136→/**
   137→ * Recorre las candidatas en orden de ranking y devuelve el índice de la
   138→ * primera FACTIBLE. Las inviables se auditan en `sink` (sin duplicados):
   139→ *   • fatiga ≥ 7 + intensidad HIGH → contraindicada (cruce con el Pilar 2)
   140→ *   • playersMin > disponibles     → inviable con la plantilla de hoy
   141→ */
   142→function feasibilityScan(
   143→  candidates: CandidateRow[],
   144→  opts: { availablePlayers: number; fatigueScore: number },
   145→  sink: DiscardedCandidate[],
   146→  seen: Set<string>,
   147→): number {
   148→  for (let c = 0; c < candidates.length; c++) {
   149→    const cand = candidates[c];
   150→    let reason: string | null = null;
   151→    if (opts.fatigueScore >= 7 && cand.intensity === "HIGH") {
   152→      reason = `Intensidad alta contraindicada con fatiga grupal ${opts.fatigueScore}/10`;
   153→    } else if (cand.playersMin > opts.availablePlayers) {
   154→      reason = `Necesita ${cand.playersMin} jugadores y solo hay ${opts.availablePlayers} disponibles`;
   155→    }
   156→    if (!reason) return c;
   157→    const key = `${cand.taskId}|${reason}`;
   158→    if (!seen.has(key)) {
   159→      seen.add(key);
   160→      sink.push({
   161→        taskId: cand.taskId,
   162→        taskTitle: cand.taskTitle,
   163→        relevance: cand.relevance,
   164→        reason,
   165→      });
   166→    }
   167→  }
   168→  return -1;
   169→}
   170→
   171→/** Orden canónico de la receta: relevancia → intensidad → título */
   172→function sortCandidates(list: CandidateRow[]): CandidateRow[] {
   173→  return list.sort(
   174→    (a, b) =>
   175→      b.relevance - a.relevance ||
   176→      (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   177→      a.taskTitle.localeCompare(b.taskTitle),
   178→  );
   179→}
   180→
   181→/**
   182→ * Candidatas de TODO el pool de la fase (todos los subprincipios visibles de
   183→ * esa fase, sin duplicar tareas). Es la red de seguridad del fallback: si el
   184→ * subprincipio exacto no tiene tarea factible, la fase sí puede tenerla.
   185→ */
   186→async function rankPhaseCandidates(
   187→  db: PipelineDb,
   188→  phase: string,
   189→): Promise<CandidateRow[]> {
   190→  const sps = await db.subPrinciple.findMany({ where: { phase: phase as GamePhase }, select: { id: true } });
   191→  if (sps.length === 0) return [];
   192→  const links = await db.taskSubPrinciple.findMany({
   193→    where: { subPrincipleId: { in: sps.map((s) => s.id) } },
   194→    include: {
   195→      task: {
   196→        select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true },
   197→      },
   198→    },
   199→  });
   200→  const byTask = new Map<string, CandidateRow>();
   201→  for (const l of links) {
   202→    const prev = byTask.get(l.task.id);
   203→    if (!prev || l.relevance > prev.relevance) {
   204→      byTask.set(l.task.id, {
   205→        taskId: l.task.id,
   206→        taskTitle: l.task.title,
   207→        relevance: l.relevance,
   208→        intensity: l.task.intensity as string,
   209→        playersMin: l.task.playersMin,
   210→        playersMax: l.task.playersMax,
   211→        durationMin: l.task.durationMin,
   212→        usageNote: l.usageNote,
   213→      });
   214→    }
   215→  }
   216→  return sortCandidates([...byTask.values()]);
   217→}
   218→
   219→/**
   220→ * Por cada hallazgo con subprincipio asociado, recorre la receta en orden de
   221→ * ranking y elige la primera tarea FACTIBLE. Si el subprincipio exacto se
   222→ * queda sin candidatas viables, amplía a la mejor tarea de la FASE completa
   223→ * (receta ampliada, auditable). El resto se audita como descartadas.
   224→ */
   225→export async function prescribeForIssues(
   226→  db: PipelineDb,
   227→  drafts: IssueDraft[],
   228→  opts: { availablePlayers: number; fatigueScore: number },
   229→): Promise<{
   230→  prescriptions: Array<{
   231→    issueIndex: number;
   232→    taskId: string;
   233→    taskTitle: string;
   234→    relevance: number;
   235→    usageNote: string | null;
   236→    durationMin: number;
   237→    rationale: string;
   238→  }>;
   239→  discarded: DiscardedCandidate[];
   240→}> {
   241→  const prescriptions: Array<{
   242→    issueIndex: number;
   243→    taskId: string;
   244→    taskTitle: string;
   245→    relevance: number;
   246→    usageNote: string | null;
   247→    durationMin: number;
   248→    rationale: string;
   249→  }> = [];
   250→  const discarded: DiscardedCandidate[] = [];
   251→  const seen = new Set<string>();
   252→  const { availablePlayers, fatigueScore } = opts;
   253→
   254→  for (let i = 0; i < drafts.length; i++) {
   255→    const draft = drafts[i];
   256→    if (!draft.subPrincipleId) continue;
   257→
   258→    let candidates = await rankTaskCandidates(db, draft.subPrincipleId);
   259→    let chosenIndex = feasibilityScan(candidates, opts, discarded, seen);
   260→    let fallbackNote = "";
   261→
   262→    // Fallback por fase: ningún subprincipio exacto tenía candidata factible
   263→    if (chosenIndex === -1 && draft.phase) {
   264→      const phaseCandidates = await rankPhaseCandidates(db, draft.phase);
   265→      chosenIndex = feasibilityScan(phaseCandidates, opts, discarded, seen);
   266→      if (chosenIndex !== -1) {
   267→        candidates = phaseCandidates;
   268→        fallbackNote = " Receta ampliada a la fase: el subprincipio exacto no tenía tarea factible.";
   269→      }
   270→    }
   271→
   272→    if (chosenIndex === -1) continue;
   273→
   274→    const chosen = candidates[chosenIndex];
   275→    // El resto de candidatas (detrás de la elegida) quedan auditadas
   276→    for (let c = chosenIndex + 1; c < candidates.length; c++) {
   277→      const cand = candidates[c];
   278→      const reason =
   279→        cand.relevance < chosen.relevance
   280→          ? `Relevancia menor en la receta (${cand.relevance} vs ${chosen.relevance})`
   281→          : "Priorizada otra candidata de mayor afinidad";
   282→      const key = `${cand.taskId}|${reason}`;
   283→      if (!seen.has(key)) {
   284→        seen.add(key);
   285→        discarded.push({
   286→          taskId: cand.taskId,
   287→          taskTitle: cand.taskTitle,
   288→          relevance: cand.relevance,
   289→          reason,
   290→        });
   291→      }
   292→    }
   293→
   294→    const intensityFit =
   295→      fatigueScore >= 7
   296→        ? "Intensidad ajustada a la fatiga del grupo."
   297→        : "";
   298→    prescriptions.push({
   299→      issueIndex: i,
   300→      taskId: chosen.taskId,
   301→      taskTitle: chosen.taskTitle,
   302→      relevance: chosen.relevance,
   303→      usageNote: chosen.usageNote,
   304→      durationMin: Math.min(chosen.durationMin, draft.kind === "PHYSICAL" ? 40 : 60),
   305→      rationale: `Entra por: ${draft.evidence}. ${chosen.usageNote ?? ""} ${intensityFit}${fallbackNote}`.trim(),
   306→    });
   307→  }
   308→
   309→  return { prescriptions, discarded };
   310→}

   316→async function recordEngineRun(
   317→  db: PipelineDb,
   318→  args: {
   319→    clubId: string;
   320→    teamId: string;
   321→    microcycleId: string | null;
   322→    trigger: "WEEKLY_PROPOSAL" | "EVIDENCE_REFRESH";
   323→    t0: number;
   324→    inputs: { pulse: { moraleScore: number; fatigueScore: number } | null; matches: Array<{ opponent: string; score: string; events: number }>; availablePlayers: number; weekStart: string };
   325→    drafts: IssueDraft[];
   326→    prescriptions: PrescriptionAudit[];
   327→    discarded: DiscardedCandidate[];
   328→  },
   329→): Promise<string> {
   330→  const run = await db.engineRun.create({
   331→    data: {
   332→      clubId: args.clubId,
   333→      teamId: args.teamId,
   334→      microcycleId: args.microcycleId,
   335→      version: ENGINE_VERSION,
   336→      trigger: args.trigger,
   337→      inputSnapshot: args.inputs as never,
   338→      issuesJson: args.drafts.map((d) => ({
   339→        kind: d.kind,
   340→        phase: d.phase ?? null,
   341→        subPrincipleId: d.subPrincipleId ?? null,
   342→        severity: d.severity,
   343→        evidence: d.evidence,
   344→      })) as never,
   345→      prescriptions: args.prescriptions as never,
   346→      discarded: args.discarded as never,
   347→      durationMs: Date.now() - args.t0,
   348→    },
   349→    select: { id: true },
   350→  });
   351→  return run.id;
   352→}

   358→export type ProposalResult = {
   359→  created: boolean;
   360→  microcycleId: string | null;
   361→  runId: string | null;
   362→  issuesCount: number;
   363→  prescriptions: PrescriptionAudit[];
   364→  discarded: DiscardedCandidate[];
   365→  durationMs: number | null;
   366→};
   367→
   368→/**
   369→ * Cocina el microciclo PROPOSED de `weekStart` desde los datos reales y
   370→ * persiste la auditoría completa en EngineRun. Si ya existe un microciclo
   371→ * para esa semana devuelve `created: false` sin tocar nada.
   372→ */
   373→export async function runWeeklyProposalPipeline(
   374→  db: PipelineDb,
   375→  args: { clubId: string; teamId: string; weekStart: Date },
   376→): Promise<ProposalResult> {
   377→  const t0 = Date.now();
   378→  const { clubId, teamId, weekStart } = args;
   379→
   380→  const existing = await db.microcycle.findUnique({
   381→    where: { teamId_weekStart: { teamId, weekStart } },
   382→    select: { id: true },
   383→  });
   384→  if (existing) {
   385→    return {
   386→      created: false,
   387→      microcycleId: existing.id,
   388→      runId: null,
   389→      issuesCount: 0,
   390→      prescriptions: [],
   391→      discarded: [],
   392→      durationMs: null,
   393→    };
   394→  }
   395→
   396→  // ---------- 1) Entradas ----------
   397→  const inputs = await collectInputs(db, teamId, weekStart);
   398→  const pulse = inputs.pulse;
   399→  const recentMatches = inputs.matches;
   400→
   401→  const availablePlayers = await db.player.count({
   402→    where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   403→  });
   404→
   405→  // ---------- 2) Diagnóstico (heurística compartida) ----------
   406→  const drafts = await buildIssueDrafts(db, teamId, inputs);
   407→  const last = recentMatches[0];
   408→
   409→  // ---------- 3) Prescripción con ranking + factibilidad ----------
   410→  const { prescriptions: rawRx, discarded } = await prescribeForIssues(db, drafts, {
   411→    availablePlayers,
   412→    fatigueScore: pulse?.fatigueScore ?? 5,
   413→  });
   414→
   415→  const auditRx: PrescriptionAudit[] = rawRx.map((p) => ({
   416→    issueEvidence: drafts[p.issueIndex].evidence,
   417→    issueKind: drafts[p.issueIndex].kind,
   418→    subPrincipleId: drafts[p.issueIndex].subPrincipleId ?? null,
   419→    taskId: p.taskId,
   420→    taskTitle: p.taskTitle,
   421→    relevance: p.relevance,
   422→    sessionLabel: "",
   423→    durationMin: p.durationMin,
   424→    rationale: p.rationale,
   425→  }));
   426→
   427→  // ---------- 4) Tareas de apoyo (MD+1 / MD-1) ----------
   428→  const recoveryTask = await db.task.findFirst({
   429→    where: { intensity: "LOW", phases: { has: "OFFENSIVE" } },
   430→    orderBy: { createdAt: "asc" },
   431→    select: { id: true, title: true },
   432→  });
   433→  const activationTask = await db.task.findFirst({
   434→    where: { intensity: { in: ["LOW", "MEDIUM"] }, phases: { has: "OFFENSIVE" } },
   435→    orderBy: { createdAt: "desc" },
   436→    select: { id: true, title: true },
   437→  });
   438→
   439→  // ---------- 5) Construcción del microciclo ----------
   440→  const tacticalDrafts = drafts.filter((d) => d.kind === "TACTICAL");
   441→  const day = (offset: number) => {
   442→    const d = new Date(weekStart);
   443→    d.setDate(d.getDate() + offset);
   444→    return d;
   445→  };
   446→
   447→  const directives = tacticalDrafts.length
   448→    ? tacticalDrafts.map((d) => `Corregir: ${d.evidence}`)
   449→    : ["Consolidar la identidad de juego de la semana"];
   450→
   451→  const summary = [
   452→    `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   453→    pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   454→    `${availablePlayers} jugadores disponibles.`,
   455→    `Prioridad de la semana: ${directives[0]}.`,
   456→  ]
   457→    .filter(Boolean)
   458→    .join(" ");
   459→
   460→  const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   461→
   462→  const microcycle = await db.microcycle.create({
   463→    data: {
   464→      clubId,
   465→      teamId,
   466→      weekStart,
   467→      status: "PROPOSED",
   468→      tacticalDirectives: directives,
   469→      methodologicalApproach: "Analítico-inductivo por proximidad, progresión a tareas globales",
   470→      diagnosisSummary: summary,
   471→      engineVersion: ENGINE_VERSION,
   472→      analyzedMatches: {
   473→        create: recentMatches.map((m) => ({ matchId: m.id })),
   474→      },
   475→      issues: {
   476→        create: drafts.map((d) => ({
   477→          subPrincipleId: d.subPrincipleId,
   478→          phase: d.phase,
   479→          kind: d.kind,
   480→          severity: d.severity,
   481→          evidence: d.evidence,
   482→          metric: d.metric as never,
   483→        })),
   484→      },
   485→      sessions: {
   486→        create: [
   487→          // MD+1 — Regeneración
   488→          {
   489→            date: day(1),
   490→            label: "MD+1",
   491→            objective: "Regeneración y asimilación del esfuerzo del partido",
   492→            durationMin: 50,
   493→            tasks: {
   494→              create: recoveryTask
   495→                ? [
   496→                    {
   497→                      order: 1,
   498→                      taskId: recoveryTask.id,
   499→                      durationMin: 40,
   500→                      rationale:
   501→                        (pulse?.fatigueScore ?? 5) >= 7
   502→                          ? "Fatiga alta: intensidad controlada para recuperar al grupo."
   503→                          : "Volumen bajo para asimilar el partido y mantener toque de balón.",
   504→                    },
   505→                  ]
   506→                : [],
   507→            },
   508→          },
   509→          // MD-3 — Sesión principal (táctica)
   510→          {
   511→            date: day(2),
   512→            label: "MD-3",
   513→            objective: tacticalDrafts[0]?.evidence ?? "Trabajo táctico principal",
   514→            durationMin: 80,
   515→            tasks: {
   516→              create: tacticalRx.slice(0, 2).map((p, idx) => ({
   517→                order: idx + 1,
   518→                taskId: p.taskId,
   519→                durationMin: p.durationMin,
   520→                rationale: p.rationale,
   521→              })),
   522→            },
   523→          },
   524→          // MD-2 — Consolidación
   525→          {
   526→            date: day(3),
   527→            label: "MD-2",
   528→            objective: "Consolidación + preparación del plan de partido",
   529→            durationMin: 70,
   530→            tasks: {
   531→              create: tacticalRx.slice(1, 2).map((p) => ({
   532→                order: 1,
   533→                taskId: p.taskId,
   534→                durationMin: p.durationMin,
   535→                rationale: p.rationale,
   536→              })),
   537→            },
   538→          },
   539→          // MD-1 — Activación pre-partido
   540→          {
   541→            date: day(4),
   542→            label: "MD-1",
   543→            objective: "Activación pre-partido: velocidad y automatismos",
   544→            durationMin: 45,
   545→            tasks: {
   546→              create: activationTask
   547→                ? [
   548→                    {
   549→                      order: 1,
   550→                      taskId: activationTask.id,
   551→                      durationMin: 35,
   552→                      rationale: "Activación neuromuscular previa al partido.",
   553→                    },
   554→                  ]
   555→                : [],
   556→            },
   557→          },
   558→        ],
   559→      },
   560→    },
   561→    include: { issues: true, sessions: { include: { tasks: true } } },
   562→  });
   563→
   564→  // ---------- 6) Trazabilidad hallazgo → tarea (prescribedIssueId) ----------
   565→  // La primera prescripción táctica entra en MD-3; la segunda aparece en MD-3
   566→  // (profundizar) Y en MD-2 (consolidar). Se enlaza el hallazgo en AMBAS.
   567→  const evidenceToIssue = new Map(microcycle.issues.map((iss) => [iss.evidence, iss.id]));
   568→  const md3 = microcycle.sessions.find((s) => s.label === "MD-3");
   569→  const md2 = microcycle.sessions.find((s) => s.label === "MD-2");
   570→  const sessionByLabel = { "MD-3": md3, "MD-2": md2 } as Record<string, typeof md3>;
   571→
   572→  for (const label of ["MD-3", "MD-2"] as const) {
   573→    const session = sessionByLabel[label];
   574→    if (!session) continue;
   575→    const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   576→    for (const p of forSession) {
   577→      const draft = drafts[p.issueIndex];
   578→      const issueId = evidenceToIssue.get(draft.evidence);
   579→      const st = session.tasks.find(
   580→        (t) => t.taskId === p.taskId && t.rationale === p.rationale,
   581→      );
   582→      if (!st) continue;
   583→      await db.sessionTask.update({
   584→        where: { id: st.id },
   585→        data: { prescribedIssueId: issueId },
   586→      });
   587→      const audit = auditRx.find(
   588→        (a) => a.taskId === p.taskId && a.issueEvidence === draft.evidence,
   589→      );
   590→      if (audit && !audit.sessionLabel) audit.sessionLabel = label;
   591→    }
   592→  }
   593→
   594→  // ---------- 7) Auditoría de la ejecución ----------
   595→  const inputSnapshot = {
   596→    weekStart: weekStart.toISOString().slice(0, 10),
   597→    pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   598→    matches: recentMatches.map((m) => ({
   599→      opponent: m.opponent,
   600→      score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   601→      events: m.report?.events.length ?? 0,
   602→    })),
   603→    availablePlayers,
   604→  };
   605→  const runId = await recordEngineRun(db, {
   606→    clubId,
   607→    teamId,
   608→    microcycleId: microcycle.id,
   609→    trigger: "WEEKLY_PROPOSAL",
   610→    t0,
   611→    inputs: inputSnapshot,
   612→    drafts,
   613→    prescriptions: auditRx,
   614→    discarded,
   615→  });
   616→
   617→  return {
   618→    created: true,
   619→    microcycleId: microcycle.id,
   620→    runId,
   621→    issuesCount: drafts.length,
   622→    prescriptions: auditRx,
   623→    discarded,
   624→    durationMs: Date.now() - t0,
   625→  };
   626→}
   627→
   628→// ============================================================================
   629→// 5) RE-PRESCRIPCIÓN EN VIVO — evidencia nueva → propuesta PROPOSED actualizada
   630→// ============================================================================
   631→
   632→export type RefreshResult = {
   633→  refreshedWeeks: string[];
   634→  evidences: string[];
   635→  runs: number;
   636→};
   637→
   638→/**
   639→ * Hook de conexión (Pilar 3 → Pilar 4). Se ejecuta al guardar una ingesta.
   640→ * Para cada microciclo aún PROPOSED de la semana del partido o la siguiente:
   641→ *   1. Regenera el diagnóstico (issues + directrices + resumen).
   642→ *   2. RE-PRESCRIBE las tareas de las sesiones principales (MD-3 / MD-2)
   643→ *      con la receta fresca y la factibilidad de hoy.
   644→ *   3. Persiste un EngineRun (trigger EVIDENCE_REFRESH) con toda la auditoría.
   645→ * Los planes ACCEPTED/ADJUSTED se respetan: nunca se tocan.
   646→ */
   647→export async function refreshProposalEvidence(
   648→  db: PipelineDb,
   649→  args: { clubId: string; teamId: string; kickoffAt: Date },
   650→): Promise<RefreshResult> {
   651→  const { clubId, teamId, kickoffAt } = args;
   652→  const base = mondayOf(kickoffAt);
   653→  const candidates = [base, addDays(base, 7)]; // misma semana y semana siguiente
   654→  const refreshedWeeks: string[] = [];
   655→  const evidences: string[] = [];
   656→  let runs = 0;
   657→
   658→  for (const weekStart of candidates) {
   659→    const t0 = Date.now();
   660→    const cycle = await db.microcycle.findUnique({
   661→      where: { teamId_weekStart: { teamId, weekStart } },
   662→      include: { sessions: { include: { tasks: true } } },
   663→    });
   664→    // Solo tocamos propuestas NO validadas por el cuerpo técnico
   665→    if (!cycle || cycle.status !== "PROPOSED") continue;
   666→
   667→    const inputs = await collectInputs(db, teamId, weekStart);
   668→    const pulse = inputs.pulse;
   669→    const drafts = await buildIssueDrafts(db, teamId, inputs);
   670→
   671→    const availablePlayers = await db.player.count({
   672→      where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   673→    });
   674→
   675→    // 1) Diagnóstico regenerado
   676→    await db.diagnosisIssue.deleteMany({
   677→      where: { microcycleId: cycle.id, status: "OPEN" },
   678→    });
   679→    await db.diagnosisIssue.createMany({
   680→      data: drafts.map((d) => ({
   681→        microcycleId: cycle.id,
   682→        subPrincipleId: d.subPrincipleId,
   683→        phase: d.phase,
   684→        kind: d.kind,
   685→        severity: d.severity,
   686→        evidence: d.evidence,
   687→        metric: d.metric as never,
   688→      })),
   689→    });
   690→
   691→    // 2) Re-prescripción con ranking + factibilidad
   692→    const { prescriptions: rawRx, discarded } = await prescribeForIssues(db, drafts, {
   693→      availablePlayers,
   694→      fatigueScore: pulse?.fatigueScore ?? 5,
   695→    });
   696→    const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   697→
   698→    // Reemplaza las tareas de MD-3 / MD-2 con la prescripción fresca
   699→    const issueRows = await db.diagnosisIssue.findMany({
   700→      where: { microcycleId: cycle.id, status: "OPEN" },
   701→      select: { id: true, evidence: true },
   702→    });
   703→    const evidenceToIssue = new Map(issueRows.map((r) => [r.evidence, r.id]));
   704→    const auditRx: PrescriptionAudit[] = [];
   705→
   706→    for (const label of ["MD-3", "MD-2"] as const) {
   707→      const session = cycle.sessions.find((s) => s.label === label);
   708→      if (!session) continue;
   709→      await db.sessionTask.deleteMany({ where: { sessionId: session.id } });
   710→
   711→      const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   712→      for (let idx = 0; idx < forSession.length; idx++) {
   713→        const p = forSession[idx];
   714→        const draft = drafts[p.issueIndex];
   715→        const issueId = evidenceToIssue.get(draft.evidence) ?? null;
   716→        await db.sessionTask.create({
   717→          data: {
   718→            sessionId: session.id,
   719→            taskId: p.taskId,
   720→            order: idx + 1,
   721→            durationMin: p.durationMin,
   722→            rationale: p.rationale,
   723→            prescribedIssueId: issueId,
   724→          },
   725→        });
   726→        auditRx.push({
   727→          issueEvidence: draft.evidence,
   728→          issueKind: draft.kind,
   729→          subPrincipleId: draft.subPrincipleId ?? null,
   730→          taskId: p.taskId,
   731→          taskTitle: p.taskTitle,
   732→          relevance: p.relevance,
   733→          sessionLabel: label,
   734→          durationMin: p.durationMin,
   735→          rationale: p.rationale,
   736→        });
   737→      }
   738→    }
   739→
   740→    // 3) Directrices + resumen legible
   741→    const last = inputs.matches[0];
   742→    const directives = tacticalRx.length
   743→      ? tacticalRx.map((p) => `Corregir: ${drafts[p.issueIndex].evidence}`)
   744→      : ["Consolidar la identidad de juego de la semana"];
   745→    await db.microcycle.update({
   746→      where: { id: cycle.id },
   747→      data: {
   748→        tacticalDirectives: directives,
   749→        diagnosisSummary: [
   750→          `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   751→          pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   752→          `Prioridad de la semana: ${directives[0]}.`,
   753→        ]
   754→          .filter(Boolean)
   755→          .join(" "),
   756→        engineVersion: ENGINE_VERSION,
   757→      },
   758→    });
   759→
   760→    // 4) Auditoría EVIDENCE_REFRESH
   761→    await recordEngineRun(db, {
   762→      clubId,
   763→      teamId,
   764→      microcycleId: cycle.id,
   765→      trigger: "EVIDENCE_REFRESH",
   766→      t0,
   767→      inputs: {
   768→        weekStart: weekStart.toISOString().slice(0, 10),
   769→        pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   770→        matches: inputs.matches.map((m) => ({
   771→          opponent: m.opponent,
   772→          score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   773→          events: m.report?.events.length ?? 0,
   774→        })),
   775→        availablePlayers,
   776→      },
   777→      drafts,
   778→      prescriptions: auditRx,
   779→      discarded,
   780→    });
   781→
   782→    runs++;
   783→    refreshedWeeks.push(weekStart.toISOString().slice(0, 10));
   784→    for (const d of drafts.slice(0, 3)) evidences.push(d.evidence);
   785→  }
   786→
   787→  return { refreshedWeeks, evidences, runs };
   788→}

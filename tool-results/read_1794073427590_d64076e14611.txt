    91→    91→// 1) RANKING DE CANDIDATAS (receta ponderada de la Biblioteca)

    93→    93→
    94→    94→/**
    95→    95→ * Devuelve todas las tareas conectadas a un subprincipio por la receta,
    96→    96→ * ordenadas por: relevancia (3 "Núcleo" primero) → intensidad suave antes que
    97→    97→ * alta → título (determinismo).
    98→    98→ */
    99→    99→export async function rankTaskCandidates(
   100→   100→  db: PipelineDb,
   101→   101→  subPrincipleId: string,
   102→   102→): Promise<
   103→   103→  Array<{
   104→   104→    taskId: string;
   105→   105→    taskTitle: string;
   106→   106→    relevance: number;
   107→   107→    intensity: string;
   108→   108→    playersMin: number;
   109→   109→    playersMax: number | null;
   110→   110→    durationMin: number;
   111→   111→    usageNote: string | null;
   112→   112→  }>
   113→   113→> {
   114→   114→  const links = await db.taskSubPrinciple.findMany({
   115→   115→    where: { subPrincipleId },
   116→   116→    include: { task: { select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true } } },
   117→   117→  });
   118→   118→  return links
   119→   119→    .map((l) => ({
   120→   120→      taskId: l.task.id,
   121→   121→      taskTitle: l.task.title,
   122→   122→      relevance: l.relevance,
   123→   123→      intensity: l.task.intensity as string,
   124→   124→      playersMin: l.task.playersMin,
   125→   125→      playersMax: l.task.playersMax,
   126→   126→      durationMin: l.task.durationMin,
   127→   127→      usageNote: l.usageNote,
   128→   128→    }))
   129→   129→    .sort(
   130→   130→      (a, b) =>
   131→   131→        b.relevance - a.relevance ||
   132→   132→        (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   133→   133→        a.taskTitle.localeCompare(b.taskTitle),
   134→   134→    );
   135→   135→}

   138→   138→// 2) PRESCRIPCIÓN CON FACTIBILIDAD — elegida + descartadas con motivo
   139→   139→// ============================================================================
   140→   140→
   141→   141→type CandidateRow = {
   142→   142→  taskId: string;
   143→   143→  taskTitle: string;
   144→   144→  relevance: number;
   145→   145→  intensity: string;
   146→   146→  playersMin: number;
   147→   147→  playersMax: number | null;
   148→   148→  durationMin: number;
   149→   149→  usageNote: string | null;
   150→   150→};
   151→   151→
   152→   152→/**
   153→   153→ * Recorre las candidatas en orden de ranking y devuelve el índice de la
   154→   154→ * primera FACTIBLE. Las inviables se auditan en `sink` (sin duplicados):
   155→   155→ *   • fatiga ≥ 7 + intensidad HIGH → contraindicada (cruce con el Pilar 2)
   156→   156→ *   • playersMin > disponibles     → inviable con la plantilla de hoy
   157→   157→ */
   158→   158→function feasibilityScan(
   159→   159→  candidates: CandidateRow[],
   160→   160→  opts: { availablePlayers: number; fatigueScore: number },
   161→   161→  sink: DiscardedCandidate[],
   162→   162→  seen: Set<string>,
   163→   163→): number {
   164→   164→  for (let c = 0; c < candidates.length; c++) {
   165→   165→    const cand = candidates[c];
   166→   166→    let reason: string | null = null;
   167→   167→    if (opts.fatigueScore >= 7 && cand.intensity === "HIGH") {
   168→   168→      reason = `Intensidad alta contraindicada con fatiga grupal ${opts.fatigueScore}/10`;
   169→   169→    } else if (cand.playersMin > opts.availablePlayers) {
   170→   170→      reason = `Necesita ${cand.playersMin} jugadores y solo hay ${opts.availablePlayers} disponibles`;
   171→   171→    }
   172→   172→    if (!reason) return c;
   173→   173→    const key = `${cand.taskId}|${reason}`;
   174→   174→    if (!seen.has(key)) {
   175→   175→      seen.add(key);
   176→   176→      sink.push({
   177→   177→        taskId: cand.taskId,
   178→   178→        taskTitle: cand.taskTitle,
   179→   179→        relevance: cand.relevance,
   180→   180→        reason,
   181→   181→      });
   182→   182→    }
   183→   183→  }
   184→   184→  return -1;
   185→   185→}
   186→   186→
   187→   187→/** Orden canónico de la receta: relevancia → intensidad → título */
   188→   188→function sortCandidates(list: CandidateRow[]): CandidateRow[] {
   189→   189→  return list.sort(
   190→   190→    (a, b) =>
   191→   191→      b.relevance - a.relevance ||
   192→   192→      (INTENSITY_ORDER[a.intensity] ?? 1) - (INTENSITY_ORDER[b.intensity] ?? 1) ||
   193→   193→      a.taskTitle.localeCompare(b.taskTitle),
   194→   194→  );
   195→   195→}
   196→   196→
   197→   197→/**
   198→   198→ * Candidatas de TODO el pool de la fase (todos los subprincipios visibles de
   199→   199→ * esa fase, sin duplicar tareas). Es la red de seguridad del fallback: si el
   200→   200→ * subprincipio exacto no tiene tarea factible, la fase sí puede tenerla.
   201→   201→ */
   202→   202→async function rankPhaseCandidates(
   203→   203→  db: PipelineDb,
   204→   204→  phase: string,
   205→   205→): Promise<CandidateRow[]> {
   206→   206→  const sps = await db.subPrinciple.findMany({
   207→   207→    where: { phase: phase as GamePhase },
   208→   208→    select: { id: true },
   209→   209→    // Determinismo: mismo pool de subprincipios en cada ejecución
   210→   210→    orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   211→   211→  });
   212→   212→  if (sps.length === 0) return [];
   213→   213→  const links = await db.taskSubPrinciple.findMany({
   214→   214→    where: { subPrincipleId: { in: sps.map((s) => s.id) } },
   215→   215→    // Determinismo del dedupe por tarea (a igual relevancia gana el primero)
   216→   216→    orderBy: [{ relevance: "desc" }, { id: "asc" }],
   217→   217→    include: {
   218→   218→      task: {
   219→   219→        select: { id: true, title: true, intensity: true, playersMin: true, playersMax: true, durationMin: true },
   220→   220→      },
   221→   221→    },
   222→   222→  });
   223→   223→  const byTask = new Map<string, CandidateRow>();
   224→   224→  for (const l of links) {
   225→   225→    const prev = byTask.get(l.task.id);
   226→   226→    if (!prev || l.relevance > prev.relevance) {
   227→   227→      byTask.set(l.task.id, {
   228→   228→        taskId: l.task.id,
   229→   229→        taskTitle: l.task.title,
   230→   230→        relevance: l.relevance,
   231→   231→        intensity: l.task.intensity as string,
   232→   232→        playersMin: l.task.playersMin,
   233→   233→        playersMax: l.task.playersMax,
   234→   234→        durationMin: l.task.durationMin,
   235→   235→        usageNote: l.usageNote,
   236→   236→      });
   237→   237→    }
   238→   238→  }
   239→   239→  return sortCandidates([...byTask.values()]);
   240→   240→}
   241→   241→
   242→   242→/**
   243→   243→ * Por cada hallazgo con subprincipio asociado, recorre la receta en orden de
   244→   244→ * ranking y elige la primera tarea FACTIBLE. Si el subprincipio exacto se
   245→   245→ * queda sin candidatas viables, amplía a la mejor tarea de la FASE completa
   246→   246→ * (receta ampliada, auditable). El resto se audita como descartadas.
   247→   247→ */
   248→   248→export async function prescribeForIssues(
   249→   249→  db: PipelineDb,
   250→   250→  drafts: IssueDraft[],
   251→   251→  opts: { availablePlayers: number; fatigueScore: number },
   252→   252→): Promise<{
   253→   253→  prescriptions: Array<{
   254→   254→    issueIndex: number;
   255→   255→    taskId: string;
   256→   256→    taskTitle: string;
   257→   257→    relevance: number;
   258→   258→    usageNote: string | null;
   259→   259→    durationMin: number;
   260→   260→    rationale: string;
   261→   261→  }>;
   262→   262→  discarded: DiscardedCandidate[];
   263→   263→}> {
   264→   264→  const prescriptions: Array<{
   265→   265→    issueIndex: number;
   266→   266→    taskId: string;
   267→   267→    taskTitle: string;
   268→   268→    relevance: number;
   269→   269→    usageNote: string | null;
   270→   270→    durationMin: number;
   271→   271→    rationale: string;
   272→   272→  }> = [];
   273→   273→  const discarded: DiscardedCandidate[] = [];
   274→   274→  const seen = new Set<string>();
   275→   275→  const { availablePlayers, fatigueScore } = opts;
   276→   276→
   277→   277→  for (let i = 0; i < drafts.length; i++) {
   278→   278→    const draft = drafts[i];
   279→   279→    if (!draft.subPrincipleId) continue;
   280→   280→
   281→   281→    let candidates = await rankTaskCandidates(db, draft.subPrincipleId);
   282→   282→    let chosenIndex = feasibilityScan(candidates, opts, discarded, seen);
   283→   283→    let fallbackNote = "";
   284→   284→
   285→   285→    // Fallback por fase: ningún subprincipio exacto tenía candidata factible
   286→   286→    if (chosenIndex === -1 && draft.phase) {
   287→   287→      const phaseCandidates = await rankPhaseCandidates(db, draft.phase);
   288→   288→      chosenIndex = feasibilityScan(phaseCandidates, opts, discarded, seen);
   289→   289→      if (chosenIndex !== -1) {
   290→   290→        candidates = phaseCandidates;
   291→   291→        fallbackNote = " Receta ampliada a la fase: el subprincipio exacto no tenía tarea factible.";
   292→   292→      }
   293→   293→    }
   294→   294→
   295→   295→    if (chosenIndex === -1) continue;
   296→   296→
   297→   297→    const chosen = candidates[chosenIndex];
   298→   298→    // El resto de candidatas (detrás de la elegida) quedan auditadas
   299→   299→    for (let c = chosenIndex + 1; c < candidates.length; c++) {
   300→   300→      const cand = candidates[c];
   301→   301→      const reason =
   302→   302→        cand.relevance < chosen.relevance
   303→   303→          ? `Relevancia menor en la receta (${cand.relevance} vs ${chosen.relevance})`
   304→   304→          : "Priorizada otra candidata de mayor afinidad";
   305→   305→      const key = `${cand.taskId}|${reason}`;
   306→   306→      if (!seen.has(key)) {
   307→   307→        seen.add(key);
   308→   308→        discarded.push({
   309→   309→          taskId: cand.taskId,
   310→   310→          taskTitle: cand.taskTitle,
   311→   311→          relevance: cand.relevance,
   312→   312→          reason,
   313→   313→        });
   314→   314→      }
   315→   315→    }
   316→   316→
   317→   317→    const intensityFit =
   318→   318→      fatigueScore >= 7
   319→   319→        ? "Intensidad ajustada a la fatiga del grupo."
   320→   320→        : "";
   321→   321→    prescriptions.push({
   322→   322→      issueIndex: i,
   323→   323→      taskId: chosen.taskId,
   324→   324→      taskTitle: chosen.taskTitle,
   325→   325→      relevance: chosen.relevance,
   326→   326→      usageNote: chosen.usageNote,
   327→   327→      durationMin: Math.min(chosen.durationMin, draft.kind === "PHYSICAL" ? 40 : 60),
   328→   328→      rationale: `Entra por: ${draft.evidence}. ${chosen.usageNote ?? ""} ${intensityFit}${fallbackNote}`.trim(),
   329→   329→    });
   330→   330→  }
   331→   331→
   332→   332→  return { prescriptions, discarded };
   333→   333→}

   380→   380→
   381→   381→export type ProposalResult = {
   382→   382→  created: boolean;
   383→   383→  microcycleId: string | null;
   384→   384→  runId: string | null;
   385→   385→  issuesCount: number;
   386→   386→  prescriptions: PrescriptionAudit[];
   387→   387→  discarded: DiscardedCandidate[];
   388→   388→  durationMs: number | null;
   389→   389→};
   390→   390→
   391→   391→/**
   392→   392→ * Cocina el microciclo PROPOSED de `weekStart` desde los datos reales y
   393→   393→ * persiste la auditoría completa en EngineRun. Si ya existe un microciclo
   394→   394→ * para esa semana devuelve `created: false` sin tocar nada.
   395→   395→ *
   396→   396→ * TODO el pipeline corre en UNA transacción interactiva: si otro request
   397→   397→ * concurrente cocina la misma semana (carrera TOCTOU), la constraint única
   398→   398→ * (teamId, weekStart) lanza P2002 → la transacción perdedora se revierte
   399→   399→ * entera y se devuelve `created: false` sin error 500 al usuario.
   400→   400→ */
   401→   401→export async function runWeeklyProposalPipeline(
   402→   402→  db: PipelineRunnerDb,
   403→   403→  args: { clubId: string; teamId: string; weekStart: Date },
   404→   404→): Promise<ProposalResult> {
   405→   405→  const t0 = Date.now();
   406→   406→  const { clubId, teamId, weekStart } = args;
   407→   407→
   408→   408→  try {
   409→   409→    return await db.$transaction(async (tx) => {
   410→   410→      const existing = await tx.microcycle.findUnique({
   411→   411→        where: { teamId_weekStart: { teamId, weekStart } },
   412→   412→        select: { id: true },
   413→   413→      });
   414→   414→      if (existing) {
   415→   415→        return {
   416→   416→          created: false,
   417→   417→          microcycleId: existing.id,
   418→   418→          runId: null,
   419→   419→          issuesCount: 0,
   420→   420→          prescriptions: [],
   421→   421→          discarded: [],
   422→   422→          durationMs: null,
   423→   423→        };
   424→   424→      }
   425→   425→
   426→   426→      // ---------- 1) Entradas ----------
   427→   427→      const inputs = await collectInputs(tx, teamId, weekStart);
   428→   428→      const pulse = inputs.pulse;
   429→   429→      const recentMatches = inputs.matches;
   430→   430→
   431→   431→      const availablePlayers = await tx.player.count({
   432→   432→        where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   433→   433→      });
   434→   434→
   435→   435→      // ---------- 2) Diagnóstico (heurística compartida) ----------
   436→   436→      const drafts = await buildIssueDrafts(tx, teamId, inputs);
   437→   437→      const last = recentMatches[0];
   438→   438→
   439→   439→      // ---------- 3) Prescripción con ranking + factibilidad ----------
   440→   440→      const { prescriptions: rawRx, discarded } = await prescribeForIssues(tx, drafts, {
   441→   441→        availablePlayers,
   442→   442→        fatigueScore: pulse?.fatigueScore ?? 5,
   443→   443→      });
   444→   444→
   445→   445→      const auditRx: PrescriptionAudit[] = rawRx.map((p) => ({
   446→   446→        issueEvidence: drafts[p.issueIndex].evidence,
   447→   447→        issueKind: drafts[p.issueIndex].kind,
   448→   448→        subPrincipleId: drafts[p.issueIndex].subPrincipleId ?? null,
   449→   449→        taskId: p.taskId,
   450→   450→        taskTitle: p.taskTitle,
   451→   451→        relevance: p.relevance,
   452→   452→        sessionLabel: "",
   453→   453→        durationMin: p.durationMin,
   454→   454→        rationale: p.rationale,
   455→   455→      }));
   456→   456→
   457→   457→      // ---------- 4) Tareas de apoyo (MD+1 / MD-1) CON FACTIBILIDAD FÍSICA ----------
   458→   458→      // La factibilidad del grupo (jugadores disponibles + umbrales de fatiga)
   459→   459→      // NO es exclusiva de la prescripción por hallazgo: las sesiones de
   460→   460→      // recuperación y activación también respetan los requisitos físicos.
   461→   461→      //   • MD+1 recuperación: SOLO intensidad LOW (regeneración) y
   462→   462→      //     playersMin ≤ disponibles.
   463→   463→      //   • MD-1 activación: LOW/MEDIUM, pero con fatiga grupal ≥ 7 se
   464→   464→      //     restringe a LOW (no añadir carga pre-partido) y playersMin ≤
   465→   465→      //     disponibles. Sin candidata factible → sesión sin tarea (la UI la
   466→   466→      //     cubre como "sesión libre"): nunca una tarea inviable en el césped.
   467→   467→      type SupportTask = { id: string; title: string; intensity: string; playersMin: number };
   468→   468→      const supportSelect = { id: true, title: true, intensity: true, playersMin: true } as const;
   469→   469→      const fatigueHigh = (pulse?.fatigueScore ?? 5) >= 7;
   470→   470→      const recoveryPool: SupportTask[] = await tx.task.findMany({
   471→   471→        where: { intensity: "LOW", phases: { has: "OFFENSIVE" } },
   472→   472→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   473→   473→        select: supportSelect,
   474→   474→      });
   475→   475→      const activationPool: SupportTask[] = await tx.task.findMany({
   476→   476→        where: { intensity: { in: ["LOW", "MEDIUM"] }, phases: { has: "OFFENSIVE" } },
   477→   477→        orderBy: [{ createdAt: "desc" }, { id: "desc" }],
   478→   478→        select: supportSelect,
   479→   479→      });
   480→   480→      const firstFeasibleSupport = (pool: SupportTask[], allowMedium: boolean): SupportTask | null =>
   481→   481→        pool.find(
   482→   482→          (t) =>
   483→   483→            t.playersMin <= availablePlayers && // plantilla suficiente hoy
   484→   484→            (!fatigueHigh || t.intensity === "LOW") && // techo de intensidad por fatiga
   485→   485→            (allowMedium || t.intensity === "LOW"), // techo de intensidad de la sesión
   486→   486→        ) ?? null;
   487→   487→      const recoveryTask = firstFeasibleSupport(recoveryPool, false);
   488→   488→      const activationTask = firstFeasibleSupport(activationPool, true);
   489→   489→
   490→   490→      // ---------- 5) Construcción del microciclo ----------
   491→   491→      const tacticalDrafts = drafts.filter((d) => d.kind === "TACTICAL");
   492→   492→      const day = (offset: number) => {
   493→   493→        const d = new Date(weekStart);
   494→   494→        d.setDate(d.getDate() + offset);
   495→   495→        return d;
   496→   496→      };
   497→   497→
   498→   498→      const directives = tacticalDrafts.length
   499→   499→        ? tacticalDrafts.map((d) => `Corregir: ${d.evidence}`)
   500→   500→        : ["Consolidar la identidad de juego de la semana"];
   501→   501→
   502→   502→      const summary = [
   503→   503→        `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   504→   504→        pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   505→   505→        `${availablePlayers} jugadores disponibles.`,
   506→   506→        `Prioridad de la semana: ${directives[0]}.`,
   507→   507→      ]
   508→   508→        .filter(Boolean)
   509→   509→        .join(" ");
   510→   510→
   511→   511→      const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   512→   512→
   513→   513→      const microcycle = await tx.microcycle.create({
   514→   514→        data: {
   515→   515→          clubId,
   516→   516→          teamId,
   517→   517→          weekStart,
   518→   518→          status: "PROPOSED",
   519→   519→          tacticalDirectives: directives,
   520→   520→          methodologicalApproach: "Analítico-inductivo por proximidad, progresión a tareas globales",
   521→   521→          diagnosisSummary: summary,
   522→   522→          engineVersion: ENGINE_VERSION,
   523→   523→          analyzedMatches: {
   524→   524→            create: recentMatches.map((m) => ({ matchId: m.id })),
   525→   525→          },
   526→   526→          issues: {
   527→   527→            create: drafts.map((d) => ({
   528→   528→              subPrincipleId: d.subPrincipleId,
   529→   529→              phase: d.phase,
   530→   530→              kind: d.kind,
   531→   531→              severity: d.severity,
   532→   532→              evidence: d.evidence,
   533→   533→              metric: d.metric as never,
   534→   534→            })),
   535→   535→          },
   536→   536→          sessions: {
   537→   537→            create: [
   538→   538→              // MD+1 — Regeneración
   539→   539→              {
   540→   540→                date: day(1),
   541→   541→                label: "MD+1",
   542→   542→                objective: "Regeneración y asimilación del esfuerzo del partido",
   543→   543→                durationMin: 50,
   544→   544→                tasks: {
   545→   545→                  create: recoveryTask
   546→   546→                    ? [
   547→   547→                        {
   548→   548→                          order: 1,
   549→   549→                          taskId: recoveryTask.id,
   550→   550→                          durationMin: 40,
   551→   551→                          rationale:
   552→   552→                            (pulse?.fatigueScore ?? 5) >= 7
   553→   553→                              ? "Fatiga alta: intensidad controlada para recuperar al grupo."
   554→   554→                              : "Volumen bajo para asimilar el partido y mantener toque de balón.",
   555→   555→                        },
   556→   556→                      ]
   557→   557→                    : [],
   558→   558→                },
   559→   559→              },
   560→   560→              // MD-3 — Sesión principal (táctica)
   561→   561→              {
   562→   562→                date: day(2),
   563→   563→                label: "MD-3",
   564→   564→                objective: tacticalDrafts[0]?.evidence ?? "Trabajo táctico principal",
   565→   565→                durationMin: 80,
   566→   566→                tasks: {
   567→   567→                  create: tacticalRx.slice(0, 2).map((p, idx) => ({
   568→   568→                    order: idx + 1,
   569→   569→                    taskId: p.taskId,
   570→   570→                    durationMin: p.durationMin,
   571→   571→                    rationale: p.rationale,
   572→   572→                  })),
   573→   573→                },
   574→   574→              },
   575→   575→              // MD-2 — Consolidación
   576→   576→              {
   577→   577→                date: day(3),
   578→   578→                label: "MD-2",
   579→   579→                objective: "Consolidación + preparación del plan de partido",
   580→   580→                durationMin: 70,
   581→   581→                tasks: {
   582→   582→                  create: tacticalRx.slice(1, 2).map((p) => ({
   583→   583→                    order: 1,
   584→   584→                    taskId: p.taskId,
   585→   585→                    durationMin: p.durationMin,
   586→   586→                    rationale: p.rationale,
   587→   587→                  })),
   588→   588→                },
   589→   589→              },
   590→   590→              // MD-1 — Activación pre-partido
   591→   591→              {
   592→   592→                date: day(4),
   593→   593→                label: "MD-1",
   594→   594→                objective: "Activación pre-partido: velocidad y automatismos",
   595→   595→                durationMin: 45,
   596→   596→                tasks: {
   597→   597→                  create: activationTask
   598→   598→                    ? [
   599→   599→                        {
   600→   600→                          order: 1,
   601→   601→                          taskId: activationTask.id,
   602→   602→                          durationMin: 35,
   603→   603→                          rationale: fatigueHigh
   604→   604→                            ? `Activación neuromuscular previa al partido: intensidad baja exigida por la fatiga grupal (${pulse?.fatigueScore}/10) y plantilla disponible (${availablePlayers}).`
   605→   605→                            : "Activación neuromuscular previa al partido, factible con la plantilla disponible.",
   606→   606→                        },
   607→   607→                      ]
   608→   608→                    : [],
   609→   609→                },
   610→   610→              },
   611→   611→            ],
   612→   612→          },
   613→   613→        },
   614→   614→        include: { issues: true, sessions: { include: { tasks: true } } },
   615→   615→      });
   616→   616→
   617→   617→      // ---------- 6) Trazabilidad hallazgo → tarea (prescribedIssueId) ----------
   618→   618→      // La primera prescripción táctica entra en MD-3; la segunda aparece en MD-3
   619→   619→      // (profundizar) Y en MD-2 (consolidar). Se enlaza el hallazgo en AMBAS.
   620→   620→      const evidenceToIssue = new Map(microcycle.issues.map((iss) => [iss.evidence, iss.id]));
   621→   621→      const md3 = microcycle.sessions.find((s) => s.label === "MD-3");
   622→   622→      const md2 = microcycle.sessions.find((s) => s.label === "MD-2");
   623→   623→      const sessionByLabel = { "MD-3": md3, "MD-2": md2 } as Record<string, typeof md3>;
   624→   624→
   625→   625→      for (const label of ["MD-3", "MD-2"] as const) {
   626→   626→        const session = sessionByLabel[label];
   627→   627→        if (!session) continue;
   628→   628→        const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   629→   629→        for (const p of forSession) {
   630→   630→          const draft = drafts[p.issueIndex];
   631→   631→          const issueId = evidenceToIssue.get(draft.evidence);
   632→   632→          const st = session.tasks.find(
   633→   633→            (t) => t.taskId === p.taskId && t.rationale === p.rationale,
   634→   634→          );
   635→   635→          if (!st) continue;
   636→   636→          await tx.sessionTask.update({
   637→   637→            where: { id: st.id },
   638→   638→            data: { prescribedIssueId: issueId },
   639→   639→          });
   640→   640→          const audit = auditRx.find(
   641→   641→            (a) => a.taskId === p.taskId && a.issueEvidence === draft.evidence,
   642→   642→          );
   643→   643→          if (audit && !audit.sessionLabel) audit.sessionLabel = label;
   644→   644→        }
   645→   645→      }
   646→   646→
   647→   647→      // ---------- 7) Auditoría de la ejecución ----------
   648→   648→      const inputSnapshot = {
   649→   649→        weekStart: weekStart.toISOString().slice(0, 10),
   650→   650→        pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   651→   651→        matches: recentMatches.map((m) => ({
   652→   652→          opponent: m.opponent,
   653→   653→          score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   654→   654→          events: m.report?.events.length ?? 0,
   655→   655→        })),
   656→   656→        availablePlayers,
   657→   657→      };
   658→   658→      const runId = await recordEngineRun(tx, {
   659→   659→        clubId,
   660→   660→        teamId,
   661→   661→        microcycleId: microcycle.id,
   662→   662→        trigger: "WEEKLY_PROPOSAL",
   663→   663→        t0,
   664→   664→        inputs: inputSnapshot,
   665→   665→        drafts,
   666→   666→        prescriptions: auditRx,
   667→   667→        discarded,
   668→   668→      });
   669→   669→
   670→   670→      return {
   671→   671→        created: true,
   672→   672→        microcycleId: microcycle.id,
   673→   673→        runId,
   674→   674→        issuesCount: drafts.length,
   675→   675→        prescriptions: auditRx,
   676→   676→        discarded,
   677→   677→        durationMs: Date.now() - t0,
   678→   678→      };
   679→   679→    }, TX_OPTS);
   680→   680→  } catch (e) {
   681→   681→    // Carrera TOCTOU: otro request cocinó la misma semana a la vez. La tx
   682→   682→    // perdedora se revierte ENTERA (sin microciclo ni EngineRun parcial) y se
   683→   683→    // responde created:false — nunca un 500 sin capturar al usuario.
   684→   684→    if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   685→   685→      const existing = await db.microcycle.findUnique({
   686→   686→        where: { teamId_weekStart: { teamId, weekStart } },
   687→   687→        select: { id: true },
   688→   688→      });
   689→   689→      return {
   690→   690→        created: false,
   691→   691→        microcycleId: existing?.id ?? null,
   692→   692→        runId: null,
   693→   693→        issuesCount: 0,
   694→   694→        prescriptions: [],
   695→   695→        discarded: [],
   696→   696→        durationMs: null,
   697→   697→      };
   698→   698→    }
   699→   699→    throw e;
   700→   700→  }
   701→   701→}
   702→   702→
   703→   703→// ============================================================================
   704→   704→// 5) RE-PRESCRIPCIÓN EN VIVO — evidencia nueva → propuesta PROPOSED actualizada
   705→   705→// ============================================================================
   706→   706→
   707→   707→export type RefreshResult = {
   708→   708→  refreshedWeeks: string[];
   709→   709→  evidences: string[];
   710→   710→  runs: number;
   711→   711→};
   712→   712→
   713→   713→/**
   714→   714→ * Hook de conexión (Pilar 3 → Pilar 4). Se ejecuta al guardar una ingesta.
   715→   715→ * Para cada microciclo aún PROPOSED de la semana del partido o la siguiente:
   716→   716→ *   1. Regenera el diagnóstico (issues + directrices + resumen).
   717→   717→ *   2. RE-PRESCRIBE las tareas de las sesiones principales (MD-3 / MD-2)
   718→   718→ *      con la receta fresca y la factibilidad de hoy.
   719→   719→ *   3. Persiste un EngineRun (trigger EVIDENCE_REFRESH) con toda la auditoría.
   720→   720→ * Los planes ACCEPTED/ADJUSTED se respetan: nunca se tocan.
   721→   721→ *
   722→   722→ * Cada semana se procesa en SU propia transacción: diagnóstico, tareas y
   723→   723→ * auditoría de esa semana se escriben todos o ninguno.
   724→   724→ */
   725→   725→export async function refreshProposalEvidence(
   726→   726→  db: PipelineRunnerDb,
   727→   727→  args: { clubId: string; teamId: string; kickoffAt: Date },
   728→   728→): Promise<RefreshResult> {
   729→   729→  const { clubId, teamId, kickoffAt } = args;
   730→   730→  const base = mondayOf(kickoffAt);
   731→   731→  const candidates = [base, addDays(base, 7)]; // misma semana y semana siguiente
   732→   732→  const refreshedWeeks: string[] = [];
   733→   733→  const evidences: string[] = [];
   734→   734→  let runs = 0;
   735→   735→
   736→   736→  for (const weekStart of candidates) {
   737→   737→    const touched = await db.$transaction(async (tx) => {
   738→   738→      const t0 = Date.now();
   739→   739→      const cycle = await tx.microcycle.findUnique({
   740→   740→        where: { teamId_weekStart: { teamId, weekStart } },
   741→   741→        include: { sessions: { include: { tasks: true } } },
   742→   742→      });
   743→   743→      // Solo tocamos propuestas NO validadas por el cuerpo técnico
   744→   744→      if (!cycle || cycle.status !== "PROPOSED") return false;
   745→   745→
   746→   746→      const inputs = await collectInputs(tx, teamId, weekStart);
   747→   747→      const pulse = inputs.pulse;
   748→   748→      const drafts = await buildIssueDrafts(tx, teamId, inputs);
   749→   749→
   750→   750→      const availablePlayers = await tx.player.count({
   751→   751→        where: { teamId, isActive: true, status: { in: ["AVAILABLE", "RETURNING"] } },
   752→   752→      });
   753→   753→
   754→   754→      // 1) Diagnóstico regenerado
   755→   755→      await tx.diagnosisIssue.deleteMany({
   756→   756→        where: { microcycleId: cycle.id, status: "OPEN" },
   757→   757→      });
   758→   758→      await tx.diagnosisIssue.createMany({
   759→   759→        data: drafts.map((d) => ({
   760→   760→          microcycleId: cycle.id,
   761→   761→          subPrincipleId: d.subPrincipleId,
   762→   762→          phase: d.phase,
   763→   763→          kind: d.kind,
   764→   764→          severity: d.severity,
   765→   765→          evidence: d.evidence,
   766→   766→          metric: d.metric as never,
   767→   767→        })),
   768→   768→      });
   769→   769→
   770→   770→      // 2) Re-prescripción con ranking + factibilidad
   771→   771→      const { prescriptions: rawRx, discarded } = await prescribeForIssues(tx, drafts, {
   772→   772→        availablePlayers,
   773→   773→        fatigueScore: pulse?.fatigueScore ?? 5,
   774→   774→      });
   775→   775→      const tacticalRx = rawRx.filter((p) => drafts[p.issueIndex].kind === "TACTICAL");
   776→   776→
   777→   777→      // Reemplaza las tareas de MD-3 / MD-2 con la prescripción fresca
   778→   778→      const issueRows = await tx.diagnosisIssue.findMany({
   779→   779→        where: { microcycleId: cycle.id, status: "OPEN" },
   780→   780→        select: { id: true, evidence: true },
   781→   781→        // Determinismo: mapeo estable evidencia → hallazgo
   782→   782→        orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   783→   783→      });
   784→   784→      const evidenceToIssue = new Map(issueRows.map((r) => [r.evidence, r.id]));
   785→   785→      const auditRx: PrescriptionAudit[] = [];
   786→   786→
   787→   787→      for (const label of ["MD-3", "MD-2"] as const) {
   788→   788→        const session = cycle.sessions.find((s) => s.label === label);
   789→   789→        if (!session) continue;
   790→   790→        await tx.sessionTask.deleteMany({ where: { sessionId: session.id } });
   791→   791→
   792→   792→        const forSession = label === "MD-3" ? tacticalRx.slice(0, 2) : tacticalRx.slice(1, 2);
   793→   793→        for (let idx = 0; idx < forSession.length; idx++) {
   794→   794→          const p = forSession[idx];
   795→   795→          const draft = drafts[p.issueIndex];
   796→   796→          const issueId = evidenceToIssue.get(draft.evidence) ?? null;
   797→   797→          await tx.sessionTask.create({
   798→   798→            data: {
   799→   799→              sessionId: session.id,
   800→   800→              taskId: p.taskId,
   801→   801→              order: idx + 1,
   802→   802→              durationMin: p.durationMin,
   803→   803→              rationale: p.rationale,
   804→   804→              prescribedIssueId: issueId,
   805→   805→            },
   806→   806→          });
   807→   807→          auditRx.push({
   808→   808→            issueEvidence: draft.evidence,
   809→   809→            issueKind: draft.kind,
   810→   810→            subPrincipleId: draft.subPrincipleId ?? null,
   811→   811→            taskId: p.taskId,
   812→   812→            taskTitle: p.taskTitle,
   813→   813→            relevance: p.relevance,
   814→   814→            sessionLabel: label,
   815→   815→            durationMin: p.durationMin,
   816→   816→            rationale: p.rationale,
   817→   817→          });
   818→   818→        }
   819→   819→      }
   820→   820→
   821→   821→      // 3) Directrices + resumen legible
   822→   822→      const last = inputs.matches[0];
   823→   823→      const directives = tacticalRx.length
   824→   824→        ? tacticalRx.map((p) => `Corregir: ${drafts[p.issueIndex].evidence}`)
   825→   825→        : ["Consolidar la identidad de juego de la semana"];
   826→   826→      await tx.microcycle.update({
   827→   827→        where: { id: cycle.id },
   828→   828→        data: {
   829→   829→          tacticalDirectives: directives,
   830→   830→          diagnosisSummary: [
   831→   831→            `Último partido: ${last ? `${last.isHome ? "casa" : "fuera"} vs ${last.opponent} (${last.goalsFor ?? 0}-${last.goalsAgainst ?? 0})` : "sin datos"}.`,
   832→   832→            pulse ? `Vestuario: ánimo ${pulse.moraleScore}/10 · fatiga ${pulse.fatigueScore}/10.` : "",
   833→   833→            `Prioridad de la semana: ${directives[0]}.`,
   834→   834→          ]
   835→   835→            .filter(Boolean)
   836→   836→            .join(" "),
   837→   837→          engineVersion: ENGINE_VERSION,
   838→   838→        },
   839→   839→      });
   840→   840→
   841→   841→      // 4) Auditoría EVIDENCE_REFRESH
   842→   842→      await recordEngineRun(tx, {
   843→   843→        clubId,
   844→   844→        teamId,
   845→   845→        microcycleId: cycle.id,
   846→   846→        trigger: "EVIDENCE_REFRESH",
   847→   847→        t0,
   848→   848→        inputs: {
   849→   849→          weekStart: weekStart.toISOString().slice(0, 10),
   850→   850→          pulse: pulse ? { moraleScore: pulse.moraleScore, fatigueScore: pulse.fatigueScore } : null,
   851→   851→          matches: inputs.matches.map((m) => ({
   852→   852→            opponent: m.opponent,
   853→   853→            score: `${m.goalsFor ?? 0}-${m.goalsAgainst ?? 0}`,
   854→   854→            events: m.report?.events.length ?? 0,
   855→   855→          })),
   856→   856→          availablePlayers,
   857→   857→        },
   858→   858→        drafts,
   859→   859→        prescriptions: auditRx,
   860→   860→        discarded,
   861→   861→      });
   862→   862→
   863→   863→      refreshedWeeks.push(weekStart.toISOString().slice(0, 10));
   864→   864→      for (const d of drafts.slice(0, 3)) evidences.push(d.evidence);
   865→   865→      return true;
   866→   866→    }, TX_OPTS);
   867→   867→
   868→   868→    if (touched) runs++;
   869→   869→  }
   870→   870→
   871→   871→  return { refreshedWeeks, evidences, runs };
   872→   872→}

    92→function GoalStepper({
    93→  label,
    94→  value,
    95→  onChange,
    96→  tone,
    97→}: {
    98→  label: string;
    99→  value: number;
   100→  onChange: (v: number) => void;
   101→  tone: "ours" | "theirs";
   102→}) {
   103→  return (
   104→    <div className={`rounded-3xl border p-4 flex flex-col items-center gap-2 ${tone === "ours" ? "border-emerald-500/40 bg-emerald-500/5" : "border-zinc-800 bg-zinc-900"}`}>
   105→      <p className="text-xs font-black tracking-widest text-zinc-400">{label}</p>
   106→      <div className="flex items-center gap-3">
   107→        <Button
   108→          type="button"
   109→          variant="outline"
   110→          size="icon"
   111→          className="h-12 w-12 rounded-2xl border-zinc-700 bg-zinc-950 text-zinc-200 hover:bg-zinc-800"
   112→          onClick={() => onChange(Math.max(0, value - 1))}
   113→          aria-label={`Restar gol ${label}`}
   114→        >
   115→          <Minus className="w-5 h-5" aria-hidden />
   116→        </Button>
   117→        <span className="w-14 text-center text-4xl font-black text-zinc-50" aria-live="polite">
   118→          {value}
   119→        </span>
   120→        <Button
   121→          type="button"
   122→          variant="outline"
   123→          size="icon"
   124→          className="h-12 w-12 rounded-2xl border-zinc-700 bg-zinc-950 text-zinc-200 hover:bg-zinc-800"
   125→          onClick={() => onChange(Math.min(20, value + 1))}
   126→          aria-label={`Sumar gol ${label}`}
   127→        >
   128→          <Plus className="w-5 h-5" aria-hidden />
   129→        </Button>
   130→      </div>
   131→    </div>
   132→  );
   133→}
   134→
   135→function SliderRow({
   136→  label,
   137→  value,
   138→  onChange,
   139→  invert,
   140→  hint,
   141→}: {
   142→  label: string;
   143→  value: number;
   144→  onChange: (v: number) => void;
   145→  invert?: boolean;
   146→  hint: string;
   147→}) {
   148→  const colorCls = invert
   149→    ? value >= 7
   150→      ? "bg-rose-500/20 border-rose-500/50 text-rose-300"
   151→      : value >= 4
   152→        ? "bg-amber-500/20 border-amber-500/50 text-amber-300"
   153→        : "bg-emerald-500/20 border-emerald-500/50 text-emerald-300"
   154→    : value <= 4
   155→      ? "bg-rose-500/20 border-rose-500/50 text-rose-300"
   156→      : value <= 6
   157→        ? "bg-amber-500/20 border-amber-500/50 text-amber-300"
   158→        : "bg-emerald-500/20 border-emerald-500/50 text-emerald-300";
   159→
   160→  return (
   161→    <div className="rounded-3xl bg-zinc-900 border border-zinc-800 px-4 py-3">
   162→      <div className="flex items-center justify-between gap-3">
   163→        <div>
   164→          <p className="text-sm font-bold text-zinc-100">{label}</p>
   165→          <p className="text-xs text-zinc-500">{hint}</p>
   166→        </div>
   167→        <span className={`min-w-[52px] text-center rounded-xl border px-2 py-1 text-lg font-black ${colorCls}`} aria-live="polite">
   168→          {value}
   169→        </span>
   170→      </div>
   171→      <Slider
   172→        value={[value]}
   173→        min={1}
   174→        max={10}
   175→        step={1}
   176→        onValueChange={(v) => onChange(v[0] ?? value)}
   177→        className="mt-3"
   178→        aria-label={label}
   179→      />
   180→    </div>
   181→  );
   182→}
   183→
   184→function RatingDots({ value }: { value: number | null }) {
   185→  return (
   186→    <span className="inline-flex items-center gap-1" aria-hidden>
   187→      {[1, 2, 3, 4, 5].map((i) => (
   188→        <span key={i} className={`w-1.5 h-1.5 rounded-full ${value && i <= value ? "bg-amber-300" : "bg-zinc-700"}`} />
   189→      ))}
   190→    </span>
   191→  );
   192→}

   194→// ---------- Flujo principal ----------
   195→
   196→export function IngestFlow({
   197→  match,
   198→  players,
   199→  initialReport,
   200→}: {
   201→  match: IngestMatchData;
   202→  players: IngestPlayerData[];
   203→  initialReport: IngestReportData | null;
   204→}) {
   205→  const router = useRouter();
   206→  const { toast } = useToast();
   207→  const [step, setStep] = useState<1 | 2 | 3 | 4>(1);
   208→  const [saving, startTransition] = useTransition();
   209→
   210→  // Paso 1
   211→  const [goalsFor, setGoalsFor] = useState(match.goalsFor ?? 0);
   212→  const [goalsAgainst, setGoalsAgainst] = useState(match.goalsAgainst ?? 0);
   213→  const [sensation, setSensation] = useState<string | null>(initialReport?.sensation ?? null);
   214→  const [load, setLoad] = useState(initialReport?.perceivedLoad ?? 5);
   215→  const [attitude, setAttitude] = useState(initialReport?.collectiveAttitude ?? 5);
   216→  const [execution, setExecution] = useState(initialReport?.tacticalExecution ?? 5);
   217→  const [globalNotes, setGlobalNotes] = useState(initialReport?.notes ?? "");
   218→
   219→  // Paso 2 (pizarra)
   220→  const [events, setEvents] = useState<EventInput[]>(initialReport?.events ?? []);
   221→  const [activeType, setActiveType] = useState<string>("BALL_LOSS");
   222→  const [activePlayerId, setActivePlayerId] = useState<string | null>(null);
   223→
   224→  // Paso 3 (cuaderno)
   225→  const [notes, setNotes] = useState<Record<string, PlayerNoteState>>(() => {
   226→    const base: Record<string, PlayerNoteState> = {};
   227→    for (const pn of initialReport?.playerNotes ?? []) {
   228→      base[pn.playerId] = { flag: pn.flag, note: pn.note ?? "" };
   229→    }
   230→    return base;
   231→  });
   232→
   233→  const [result, setResult] = useState<SubmitResult | null>(null);
   234→
   235→  const counts = useMemo(() => {
   236→    const map: Record<string, number> = {};
   237→    for (const e of events) map[e.type] = (map[e.type] ?? 0) + 1;
   238→    return map;
   239→  }, [events]);
   240→
   241→  const notesFilled = useMemo(
   242→    () => Object.values(notes).filter((n) => n.flag !== null || n.note.trim() !== "").length,
   243→    [notes],
   244→  );
   245→
   246→  function addEvent(x: number, y: number) {
   247→    setEvents((prev) => [
   248→      ...prev,
   249→      { x, y, type: activeType, playerId: activePlayerId, minute: null },
   250→    ]);
   251→  }
   252→
   253→  function removeEvent(index: number) {
   254→    setEvents((prev) => prev.filter((_, i) => i !== index));
   255→  }
   256→
   257→  function save() {
   258→    const playerNotes: NoteInput[] = [];
   259→    for (const [playerId, n] of Object.entries(notes)) {
   260→      if (n.flag !== null || n.note.trim() !== "") {
   261→        playerNotes.push({ playerId, flag: n.flag ?? "GREEN", note: n.note.trim() || null });
   262→      }
   263→    }
   264→    startTransition(async () => {
   265→      const res = await submitMatchReport({
   266→        gameId: match.id,
   267→        goalsFor,
   268→        goalsAgainst,
   269→        sensation: sensation ?? "AMBER",
   270→        perceivedLoad: load,
   271→        collectiveAttitude: attitude,
   272→        tacticalExecution: execution,
   273→        notes: globalNotes.trim() || null,
   274→        events,
   275→        playerNotes,
   276→      });
   277→      if (res.ok) {
   278→        setResult(res);
   279→        setStep(4);
   280→        router.refresh();
   281→      } else {
   282→        toast({ title: "No se pudo guardar", description: res.error, variant: "destructive", duration: 5000 });
   283→      }
   284→    });
   285→  }
   286→
   287→  const kickoffLabel = format(new Date(match.kickoffAt), "EEE d MMM · HH:mm'h'", { locale: es });
   288→
   289→  return (
   290→    <div>
   291→      {/* ---------- Cabecera ---------- */}
   292→      <header className="sticky top-0 z-20 border-b border-zinc-800 bg-zinc-950/90 backdrop-blur">
   293→        <div className="mx-auto max-w-5xl px-4 py-4 flex items-center justify-between gap-3">
   294→          <div className="min-w-0">
   295→            <Link href="/dashboard/matches" className="inline-flex items-center gap-1 text-xs text-zinc-500 hover:text-zinc-300 min-h-[44px]">
   296→              <ArrowLeft className="w-3.5 h-3.5" aria-hidden /> Partidos
   297→            </Link>
   298→            <h1 className="text-xl font-black text-zinc-50 leading-tight truncate">
   299→              {match.isHome ? "vs" : "@"} {match.opponent}
   300→            </h1>
   301→            <p className="text-xs text-zinc-500">
   302→              {kickoffLabel}
   303→              {match.competition ? ` · ${match.competition}` : ""}
   304→            </p>
   305→          </div>
   306→          {match.status === "ANALYZED" && (
   307→            <Badge variant="outline" className="rounded-full bg-zinc-800 text-zinc-300 border-zinc-700 shrink-0">
   308→              Editando ingesta
   309→            </Badge>
   310→          )}
   311→        </div>
   312→        {/* Indicador de pasos */}
   313→        <div className="mx-auto max-w-5xl px-4 pb-3">
   314→          <div className="grid grid-cols-3 gap-2">
   315→            {STEPS.map((s) => {
   316→              const done = step > s.n;
   317→              const active = step === s.n;
   318→              return (
   319→                <div
   320→                  key={s.n}
   321→                  className={`rounded-2xl border px-3 py-2 flex items-center gap-2 transition ${
   322→                    active
   323→                      ? "border-emerald-500/50 bg-emerald-500/10"
   324→                      : done
   325→                        ? "border-emerald-500/30 bg-emerald-500/5"
   326→                        : "border-zinc-800 bg-zinc-900"
   327→                  }`}
   328→                >
   329→                  <span className={`flex h-6 w-6 items-center justify-center rounded-full text-xs font-black ${active ? "bg-emerald-500 text-zinc-950" : done ? "bg-emerald-500/30 text-emerald-300" : "bg-zinc-800 text-zinc-500"}`}>
   330→                    {done ? "✓" : s.n}
   331→                  </span>
   332→                  <span className={`text-xs font-bold ${active ? "text-emerald-300" : done ? "text-emerald-400/70" : "text-zinc-500"}`}>
   333→                    {s.label}
   334→                  </span>
   335→                </div>
   336→              );
   337→            })}
   338→          </div>
   339→        </div>
   340→      </header>
   341→
   342→      <div className="mx-auto max-w-5xl px-4 pt-5">
   343→        {/* ============================================================ */}
   344→        {/* PASO 1 — SEMÁFORO + SLIDERS                                  */}
   345→        {/* ============================================================ */}
   346→        {step === 1 && (
   347→          <section aria-label="Paso 1: semáforo y carga" className="space-y-4">
   348→            <div className="grid grid-cols-2 gap-3">
   349→              <GoalStepper label="NOSOTROS" value={goalsFor} onChange={setGoalsFor} tone="ours" />
   350→              <GoalStepper label={match.opponent.toUpperCase()} value={goalsAgainst} onChange={setGoalsAgainst} tone="theirs" />
   351→            </div>
   352→
   353→            <p className="text-sm text-zinc-400">
   354→              ¿Cómo quedó el grupo? <span className="text-zinc-600">Un toque:</span>
   355→            </p>
   356→            <div className="grid grid-cols-3 gap-2">
   357→              {SENSATIONS.map((s) => {
   358→                const meta = SEMAPHORE_META[s];
   359→                const selected = sensation === s;
   360→                return (
   361→                  <button
   362→                    key={s}
   363→                    type="button"
   364→                    onClick={() => setSensation(s)}
   365→                    aria-pressed={selected}
   366→                    className={`min-h-[76px] rounded-3xl border px-3 py-3 flex flex-col items-center justify-center gap-1.5 transition ${selected ? meta.btn : "bg-zinc-900 border-zinc-800 text-zinc-400"}`}
   367→                  >
   368→                    <span className={`w-5 h-5 rounded-full ${meta.dot}`} aria-hidden />
   369→                    <span className="text-sm font-black">{meta.label}</span>
   370→                    <span className="text-[11px] text-zinc-500 text-center leading-tight">{meta.hint}</span>
   371→                  </button>
   372→                );
   373→              })}
   374→            </div>
   375→
   376→            <div className="space-y-3">
   377→              <SliderRow label="Carga percibida" hint="Cuánto pesó el partido al grupo" value={load} onChange={setLoad} invert />
   378→              <SliderRow label="Actitud colectiva" hint="Compromiso y energía del vestuario" value={attitude} onChange={setAttitude} />
   379→              <SliderRow label="Ejecución táctica" hint="Cuánto de lo planificado salió" value={execution} onChange={setExecution} />
   380→            </div>
   381→
   382→            <Input
   383→              className="min-h-[48px] rounded-2xl bg-zinc-900 border-zinc-800 text-zinc-100 placeholder:text-zinc-500"
   384→              placeholder="Nota global opcional (contexto rival, bajas…)"
   385→              maxLength={500}
   386→              value={globalNotes}
   387→              onChange={(e) => setGlobalNotes(e.target.value)}
   388→            />
   389→
   390→            <Button
   391→              onClick={() => setStep(2)}
   392→              disabled={sensation === null}
   393→              className="w-full min-h-[60px] rounded-3xl bg-emerald-500 hover:bg-emerald-400 text-zinc-950 font-black text-base"
   394→            >
   395→              Siguiente: pizarra táctica <ArrowRight className="w-5 h-5" aria-hidden />
   396→            </Button>
   397→            {sensation === null && (
   398→              <p className="text-center text-xs text-zinc-600">Elige el semáforo global para continuar</p>
   399→            )}
   400→          </section>
   401→        )}
   402→
   403→        {/* ============================================================ */}
   404→        {/* PASO 2 — PIZARRA TÁCTIL                                      */}
   405→        {/* ============================================================ */}
   406→        {step === 2 && (
   407→          <section aria-label="Paso 2: pizarra táctica" className="space-y-4">
   408→            <p className="text-sm text-zinc-400">
   409→              Elige el tipo de evento y <span className="font-bold text-zinc-200">toca el campo</span> donde ocurrió.
   410→              Toca un punto para borrarlo.
   411→            </p>
   412→
   413→            {/* Paleta de eventos */}
   414→            <div className="grid grid-cols-4 gap-2">
   415→              {QUICK_EVENT_TYPES.map((type) => {
   416→                const meta = EVENT_META[type];
   417→                const Icon = meta.icon;
   418→                const selected = activeType === type;
   419→                return (
   420→                  <button
   421→                    key={type}
   422→                    type="button"
   423→                    onClick={() => setActiveType(type)}
   424→                    aria-pressed={selected}
   425→                    className={`relative min-h-[68px] rounded-2xl border px-1.5 py-2 flex flex-col items-center justify-center gap-1 transition ${selected ? meta.btn : "bg-zinc-900 border-zinc-800 text-zinc-400"}`}
   426→                  >
   427→                    <Icon className="w-5 h-5" aria-hidden />
   428→                    <span className="text-[11px] font-bold leading-tight text-center">{meta.label}</span>
   429→                    {(counts[type] ?? 0) > 0 && (
   430→                      <span className={`absolute -top-1.5 -right-1.5 min-w-[20px] h-5 px-1 rounded-full text-[11px] font-black flex items-center justify-center ${meta.text} bg-zinc-950 border border-zinc-700`}>
   431→                        {counts[type]}
   432→                      </span>
   433→                    )}
   434→                  </button>
   435→                );
   436→              })}
   437→            </div>
   438→
   439→            {/* Responsable del evento */}
   440→            <div>
   441→              <p className="text-xs font-bold text-zinc-500 uppercase tracking-widest mb-2">
   442→                Responsable {activePlayerId ? "(toca «Sin responsable» para quitar)" : "(opcional)"}
   443→              </p>
   444→              <div className="flex gap-2 overflow-x-auto pb-1 [scrollbar-width:none] [-ms-overflow-style:none] [&::-webkit-scrollbar]:hidden">
   445→                <button
   446→                  type="button"
   447→                  onClick={() => setActivePlayerId(null)}
   448→                  aria-pressed={activePlayerId === null}
   449→                  className={`shrink-0 min-h-[44px] px-4 py-2 rounded-full border text-sm font-bold transition ${activePlayerId === null ? "bg-zinc-700 border-zinc-500 text-zinc-100" : "bg-zinc-900 border-zinc-800 text-zinc-400"}`}
   450→                >
   451→                  Sin responsable
   452→                </button>
   453→                {players.map((p) => {
   454→                  const selected = activePlayerId === p.id;
   455→                  return (
   456→                    <button
   457→                      key={p.id}
   458→                      type="button"
   459→                      onClick={() => setActivePlayerId(selected ? null : p.id)}
   460→                      aria-pressed={selected}
   461→                      className={`shrink-0 min-h-[44px] px-3.5 py-2 rounded-full border text-sm font-bold transition ${selected ? "bg-emerald-500/15 border-emerald-500/50 text-emerald-300" : "bg-zinc-900 border-zinc-800 text-zinc-400"}`}
   462→                    >
   463→                      <span className="font-black mr-1">{p.shirtNumber ?? "–"}</span>
   464→                      {p.lastName}
   465→                    </button>
   466→                  );
   467→                })}
   468→              </div>
   469→            </div>
   470→
   471→            {/* Campo táctil */}
   472→            <div
   473→              className="relative rounded-3xl overflow-hidden border border-zinc-800 select-none"
   474→              role="application"
   475→              aria-label="Pizarra táctica: toca para marcar eventos. Campo propio a la izquierda."
   476→            >
   477→              <svg
   478→                viewBox="0 0 150 100"
   479→                preserveAspectRatio="none"
   480→                className="block w-full aspect-[3/2] touch-none"
   481→                onClick={(e) => {
   482→                  const rect = e.currentTarget.getBoundingClientRect();
   483→                  const x = Math.min(100, Math.max(0, ((e.clientX - rect.left) / rect.width) * 100));
   484→                  const y = Math.min(100, Math.max(0, ((e.clientY - rect.top) / rect.height) * 100));
   485→                  addEvent(Math.round(x * 10) / 10, Math.round(y * 10) / 10);
   486→                }}
   487→              >
   488→                {/* césped */}
   489→                <rect width="150" height="100" fill="#06231a" />
   490→                {/* líneas */}
   491→                <g stroke="#14532d" strokeWidth="0.8" fill="none">
   492→                  <rect x="2" y="2" width="146" height="96" />
   493→                  <line x1="75" y1="2" x2="75" y2="98" />
   494→                  <circle cx="75" cy="50" r="13" />
   495→                  {/* área izquierda */}
   496→                  <rect x="2" y="25" width="18" height="50" />
   497→                  <rect x="2" y="38" width="8" height="24" />
   498→                  <circle cx="13" cy="50" r="0.9" fill="#14532d" />
   499→                  {/* área derecha */}
